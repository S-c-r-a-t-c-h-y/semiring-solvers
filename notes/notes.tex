For convenience, we recall some definitions we'll be using.

\subsection{Algebraic Structures}

\begin{definition}[Semigroup]
  A \emph{semigroup} is a set $S$ equipped with an associative binary operation $\cdot : S \times S \to S$ such that for all $a, b, c \in S$:
  \begin{itemize}
    \item $a \cdot (b \cdot c) = (a \cdot b) \cdot c$ (associativity)
  \end{itemize}
\end{definition}

\begin{definition}[Monoid]
  A \emph{monoid} is a set $M$ equipped with an associative binary operation $\cdot : M \times M \to M$ and an identity element $e \in M$ such that for all $a, b, c \in M$:
  \begin{itemize}
    \item $a \cdot (b \cdot c) = (a \cdot b) \cdot c$ (associativity)
    \item $e \cdot a = a \cdot e = a$ (identity element)
  \end{itemize}
\end{definition}

\begin{definition}[Group]
  A \emph{group} is a set $G$ equipped with an associative binary operation $\cdot : G \times G \to G$, an identity element $e \in G$, and an inverse operation such that for all $a, b, c \in G$:
  \begin{itemize}
    \item $a \cdot (b \cdot c) = (a \cdot b) \cdot c$ (associativity)
    \item $e \cdot a = a \cdot e = a$ (identity element)
    \item For each $a \in G$, there exists an element $a^{-1} \in G$ such that $a \cdot a^{-1} = a^{-1} \cdot a = e$ (inverse element)
  \end{itemize}
A group is called \emph{abelian} if its operation is commutative, i.e., for all $a, b \in G$, $a \cdot b = b \cdot a$.
\end{definition}

\begin{definition}[Semiring]
  A \emph{semiring} is a set $R$ equipped with two binary operations: addition $+$ and multiplication $\cdot$, such that:
  \begin{itemize}
    \item $(R, +)$ is a commutative monoid with identity element $0$.
    \item $(R, \cdot)$ is a monoid with identity element $1$.
    \item Multiplication distributes over addition from both sides: for all $a, b, c \in R$,
      \begin{align*}
        a \cdot (b + c) &= a \cdot b + a \cdot c, \\
        (a + b) \cdot c &= a \cdot c + b \cdot c.
      \end{align*}
    \item The additive identity $0$ is an absorbing element for multiplication: for all $a \in R$, $0 \cdot a = a \cdot 0 = 0$.
  \end{itemize}
\end{definition}


\subsection{Categories}

\begin{definition}[Category]
  A \emph{category} $\mathcal{C}$ consists of:
  \begin{itemize}
    \item A class of objects $\text{Ob}(\mathcal{C})$,
    \item For each pair of objects $a, b \in \text{Ob}(\mathcal{C})$, a set of morphisms $\text{Hom}_{\mathcal{C}}(a, b)$,
    \item For each object $a \in \text{Ob}(\mathcal{C})$, an identity morphism $\text{id}_a \in \text{Hom}_{\mathcal{C}}(a, a)$,
    \item A composition operation $\circ$ such that for morphisms $f: a \to b$ and $g: b \to c$, there is a morphism $g \circ f: a \to c$,
  \end{itemize}
  satisfying the following axioms:
  \begin{itemize}
    \item Associativity: For morphisms $f: a \to b$, $g: b \to c$, and $h: c \to d$, we have $h \circ (g \circ f) = (h \circ g) \circ f$.
    \item Identity: For any morphism $f: a \to b$, we have $\text{id}_b \circ f = f = f \circ \text{id}_a$.
  \end{itemize}
\end{definition}

\begin{example}[Set Category]
  The category $\mathbf{Set}$ has sets as objects and functions as morphisms. The identity morphism for a set $a$ is the identity function $\text{id}_a: a \to a$ defined by $\text{id}_a(x) = x$ for all $x \in a$. Composition of morphisms is given by function composition.
\end{example}

\begin{example}[Walking Morphism]
  The \emph{walking morphism} category $\mathbf{2}$ or $I$ has two objects, say $0$ and $1$, and a single non-identity morphism $f: 0 \to 1$. The identity morphisms are $\text{id}_0: 0 \to 0$ and $\text{id}_1: 1 \to 1$. The composition is defined by the identities and the morphism $f$.
\end{example}

\begin{definition}[Product]
  Given two objects $a$ and $b$ in a category $\mathcal{C}$, a \emph{product} of $a$ and $b$ is an object $a \times b$ together with two morphisms $\pi_a: a \times b \to a$ and $\pi_b: a \times b \to b$ such that for any object $c$ and morphisms $f: c \to a$ and $g: c \to b$, there exists a unique morphism $\langle f, g \rangle: c \to a \times b$ making the following diagram commute:
  $$
  \begin{tikzcd}
    & c \arrow[dl, "f"'] \arrow[dr, "g"] \arrow[dashed, d, "{\langle f, g \rangle}"] & \\
    a & a \times b \arrow[l, "\pi_a"] \arrow[r, "\pi_b"'] & b
  \end{tikzcd}
  $$
\end{definition}

\begin{definition}[Coproduct]
  Given two objects $a$ and $b$ in a category $\mathcal{C}$, a \emph{coproduct} of $a$ and $b$ is an object $a + b$ together with two morphisms $\iota_a: a \to a + b$ and $\iota_b: b \to a + b$ such that for any object $c$ and morphisms $f: a \to c$ and $g: b \to c$, there exists a unique morphism $[f, g]: a + b \to c$ making the following diagram commute:
  $$
  \begin{tikzcd}
    a \arrow[r, "\iota_a"] \arrow[dr, "f"'] & a + b \arrow[dashed, d, "{[f, g]}"] & b \arrow[l, "\iota_b"'] \arrow[dl, "g"] \\
    & c &
  \end{tikzcd}
  $$
\end{definition}

\begin{definition}[Monoidal Category]
  A \emph{monoidal category} is a category $\mathcal{C}$ equipped with a bifunctor $\otimes: \mathcal{C} \times \mathcal{C} \to \mathcal{C}$, an object $I$ called the unit object, and natural isomorphisms (called associators and unitors) that satisfy certain coherence conditions. Specifically, for all objects $a, b, c$ in $\mathcal{C}$, there are natural isomorphisms:
  \begin{itemize}
    \item Associator: $\alpha_{a,b,c}: (a \otimes b) \otimes c \to a \otimes (b \otimes c)$,
    \item Left unitor: $\lambda_a: I \otimes a \to a$,
    \item Right unitor: $\rho_a: a \otimes I \to a$,
  \end{itemize}
  such that the following diagrams commute:
  $$
  \begin{tikzcd}
    ((a \otimes b) \otimes c) \otimes d \arrow[r, "\alpha_{a,b,c} \otimes 1_d"] & (a \otimes (b \otimes c)) \otimes d \arrow[r, "\alpha_{a,b\otimes c,d}"] & a \otimes ((b \otimes c) \otimes d) \\
    (a \otimes b) \otimes (c \otimes d) \arrow[u, "\alpha_{a,b,c\otimes d}"] & & a \otimes (b \otimes (c \otimes d)) \arrow[u, "1_a \otimes \alpha_{b,c,d}"]
  \end{tikzcd}
  $$
  $$
  \begin{tikzcd}
    (I \otimes a) \otimes b \arrow[r, "\alpha_{I,a,b}"] & I \otimes (a \otimes b) & a = a \otimes I \arrow[l, "\rho_a"'] \arrow[r, "\lambda_a"] & a \otimes I \arrow[r, "\alpha_{a,I,I}"] & (a \otimes I) \otimes I
  \end{tikzcd}
  $$
\end{definition}


\begin{definition}[Functor]
  Given two categories $\mathcal{C}$ and $\mathcal{D}$, a \emph{functor} $F: \mathcal{C} \to \mathcal{D}$ consists of:
  \begin{itemize}
    \item A mapping $F: \text{Ob}(\mathcal{C}) \to \text{Ob}(\mathcal{D})$ that assigns to each object $a$ in $\mathcal{C}$ an object $F(a)$ in $\mathcal{D}$,
    \item For each pair of objects $a, b \in \text{Ob}(\mathcal{C})$, a mapping $F: \text{Hom}_{\mathcal{C}}(a, b) \to \text{Hom}_{\mathcal{D}}(F(a), F(b))$ that assigns to each morphism $f: a \to b$ in $\mathcal{C}$ a morphism $F(f): F(a) \to F(b)$ in $\mathcal{D}$,
  \end{itemize}
  such that for all objects $a, b, c$ in $\mathcal{C}$ and morphisms $f: a \to b$, $g: b \to c$, the following conditions hold:
  \begin{itemize}
    \item $F(g \circ f) = F(g) \circ F(f)$ (preservation of composition),
    \item $F(\text{id}_a) = \text{id}_{F(a)}$ (preservation of identity).
  \end{itemize}
  For a category $\mathcal{C}$, the \emph{identity functor} $\text{Id}_{\mathcal{C}}: \mathcal{C} \to \mathcal{C}$ is defined by $\text{Id}_{\mathcal{C}}(a) = a$ for all objects $a$ and $\text{Id}_{\mathcal{C}}(f) = f$ for all morphisms $f$.

  \medskip

  \begin{itemize}
    \item A functor $F: \mathcal{C} \to \mathcal{D}$ is \emph{faithful} if, for every pair of objects $a, b \in \text{Ob}(\mathcal{C})$, the map $F: \text{Hom}_{\mathcal{C}}(a, b) \to \text{Hom}_{\mathcal{D}}(Fa, Fb)$ is injective.
    \item A functor $F: \mathcal{C} \to \mathcal{D}$ is \emph{full} if, for every pair of objects $a, b \in \text{Ob}(\mathcal{C})$, the map $F: \text{Hom}_{\mathcal{C}}(a, b) \to \text{Hom}_{\mathcal{D}}(Fa, Fb)$ is surjective.
    \item A functor that is both full and faithful is called \emph{fully faithful}.
  \end{itemize}

\end{definition}

\textbf{Notation:} We often denote the image of an object $a$ under a functor $F$ as $Fa$ instead of $F(a)$ and the image of a morphism $f: a \to b$ as $Ff: Fa \to Fb$.

\begin{example}[Diagonal functor]
  Given a category $\mathcal{C}$, the \emph{diagonal functor} $\Delta: \mathcal{C} \to \mathcal{C} \times \mathcal{C}$ is defined by $\Delta(a) = (a, a)$ for each object $a$ in $\mathcal{C}$ and $\Delta(f) = (f, f)$ for each morphism $f: a \to b$ in $\mathcal{C}$.
\end{example}

\begin{definition}[Product preserving functor]
  A functor $F: \mathcal{C} \to \mathcal{D}$ is said to \emph{preserve products} if for every collection of objects $\{a_i\}_{i \in I}$ in $\mathcal{C}$, the natural morphism from $F(\prod_{i \in I} a_i)$ to $\prod_{i \in I} F(a_i)$ is an isomorphism in $\mathcal{D}$. In other words, $F$ preserves the structure of products up to isomorphism.
\end{definition}

\begin{definition}[Natural Transformation]
  Given two functors $F, G: \mathcal{C} \to \mathcal{D}$ between categories $\mathcal{C}$ and $\mathcal{D}$, a \emph{natural transformation} $\eta: F \to G$ consists of a family of morphisms $\{\eta_a: Fa \to Ga\}_{a \in \text{Ob}(\mathcal{C})}$ such that for every morphism $f: a \to b$ in $\mathcal{C}$, the following diagram commutes:
  \[
  \begin{tikzcd}
    Fa \arrow[r, "Ff"] \arrow[d, "\eta_a"'] & Fb \arrow[d, "\eta_b"] \\
    Ga \arrow[r, "Gf"'] & Gb
  \end{tikzcd}
  \]
\end{definition}


\begin{definition}[Adjunction]
  Given two categories $\mathcal{C}$ and $\mathcal{D}$, a functor $F: \mathcal{C} \to \mathcal{D}$ is a \emph{left adjoint} to a functor $G: \mathcal{D} \to \mathcal{C}$ (and $G$ is a \emph{right adjoint} to $F$) if there is a natural isomorphism:
  \[
  \text{Hom}_{\mathcal{D}}(Fa, b) \cong \text{Hom}_{\mathcal{C}}(a, Gb)
  \]
  for all objects $a$ in $\mathcal{C}$ and $b$ in $\mathcal{D}$. This means that for every morphism $f: Fa \to b$ in $\mathcal{D}$, there is a unique morphism $g: a \to Gb$ in $\mathcal{C}$ such that the following diagram commutes:
  \[
  \begin{tikzcd}
    Fa \arrow[r, "f"] & b \arrow[d, "G"] \\
    a \arrow[u, "F"'] \arrow[r, "g"'] & Gb
  \end{tikzcd}
  \]
  We often denote this adjunction by $F \dashv G$.
\end{definition}

One important definition is that of the universal arrow from MacLane's \textit{Categories for the Working Mathematician}~\cite{maclane:71}.

\begin{definition}[Universal arrow]
  Let $U : A \to C$ be a functor, and let $c \in C$.

  \medskip

  An \emph{universal arrow} from $c$ to $U$ is a pair $(F c, \eta_c)$ where:
  \begin{itemize}
    \item $F c \in A$
    \item $\eta_c : c \to UFc$ is a morphism in $C$
  \end{itemize}
  such that for every $a \in A$, and every $f : c \to Ua$ in $C$, there exists a unique morphism $f^\dagger : F c \to a$ in $A$ such that the following diagram commutes:
  \[
  \begin{tikzcd}
    c \arrow[r, "\eta_c"] \arrow[dr, "f"'] & UFc \arrow[dashed, d, "U{f^\dagger}"] \\
    & Ua
  \end{tikzcd}
  \quad\quad
  \begin{tikzcd}
    F c \arrow[d, "f^\dagger"]\\
    a
  \end{tikzcd}
  \]
  We can dualize this definition to get the notion of a universal arrow from $U$ to $c$.

  \medskip

  A universal arrow from $U$ to $c$ is a pair $(F c, \epsilon_c)$ where:
  \begin{itemize}
    \item $F c \in A$
    \item $\epsilon_c : UFc \to c$ is a morphism in $C$
  \end{itemize}
  such that for every $a \in A$, and every $f : Ua \to c$ in $C$, there exists a unique morphism $f^\dagger : a \to F c$ in $A$ such that the following diagram commutes:
  \[
  \begin{tikzcd}
    c & UFc \arrow[l, "\epsilon_c"']  \\
    & Ua \arrow[dashed, u, "U{f^\dagger}"'] \arrow[ul, "f"]
  \end{tikzcd}
  \quad\quad
  \begin{tikzcd}
    F c\\
    a \arrow[u, "f^\dagger"']
  \end{tikzcd}
  \]
\end{definition}

\begin{example}[Adjunction from Universal Arrows]
  Given a functor $U : A \to C$ such that for every object $c \in C$, there exists a universal arrow from $c$ to $U$, we can construct an adjunction $F \dashv U$ where:
  \begin{itemize}
    \item The functor $F : C \to A$ is defined by sending each object $c$ in $C$ to the object $Fc$ in $A$ that appears in the universal arrow from $c$ to $U$, and sending each morphism $f: c \to c'$ in $C$ to the unique morphism $Ff: Fc \to Fc'$ in $A$ that makes the corresponding diagram commute.
    \item The unit $\eta: \text{Id}_C \to UF$ is given by the family of morphisms $\{\eta_c: c \to UFc\}_{c \in C}$ that are part of the universal arrows.
  \end{itemize}
\end{example}

Universal arrows have more uses than just defining adjunctions. Here are other examples.

\begin{example}[Defining the coproduct with an universal arrow]
  Let $A$ be a category, and let $a, b \in A$. A \emph{coproduct} of $a$ and $b$ is a universal arrow from $(a, b)$ to the diagonal functor $\Delta: A \to A \times A$, where $\Delta(c) = (c, c)$ for any object $c \in A$.
  $$
  \begin{tikzcd}
    a \arrow[r, "\iota_a"] \arrow[dr, "f"'] & a + b \arrow[dashed, d, "{[f, g]}"] & b \arrow[l, "\iota_b"'] \arrow[dl, "g"] \\
    & c &
  \end{tikzcd}
  \quad\sim\quad
  \begin{tikzcd}
    (a, b) \arrow[r, "\eta_{(a, b)}"] \arrow[dr, "{(f, g)}"'] & \Delta(a + b) \arrow[dashed, d, "\Delta{[f, g]}"] \\
    & \Delta c
  \end{tikzcd}
  $$
  We have $\eta_{(a, b)} = (\iota_a, \iota_b)$.
\end{example}

\begin{example}[Defining the product with an universal arrow]
  Let $A$ be a category, and let $a, b \in A$. A \emph{product} of $a$ and $b$ is a universal arrow from the diagonal functor $\Delta: A \to A \times A$ to $(a, b)$.
  $$
  \begin{tikzcd}
    & c \arrow[dl, "f"'] \arrow[dr, "g"] \arrow[dashed, d, "{\langle f, g \rangle}"] & \\
    a & a \times b \arrow[l, "\pi_a"] \arrow[r, "\pi_b"'] & b
  \end{tikzcd}
  \quad\sim\quad
  \begin{tikzcd}
    \Delta c \arrow[r, "\epsilon_{(a, b)}"] \arrow[dr, "{(f, g)}"'] & \Delta (a \times b) \arrow[dashed, d, "{(\pi_a, \pi_b)}"] \\
    & (a, b)
  \end{tikzcd}
  $$
  We have $\epsilon_{(a, b)} = \langle f, g\rangle$.
\end{example}



\begin{definition}[Pullback of two morphisms]
  Let $f : X \to Z$ and $g : Y \to Z$ be two morphisms in a category $\mathcal{C}$. A \emph{pullback} of $f$ and $g$ consists of an object $P$ and two morphisms $p_1 : P \to X$ and $p_2 : P \to Y$ such that the following diagram commutes:
  \[
  \begin{tikzcd}
    P \arrow[r, "p_2"] \arrow[d, "p_1"'] & Y \arrow[d, "g"]\\
    X \arrow[r, "f"'] & Z
  \end{tikzcd}
  \]
  Moreover, for any triple $(Q, q_1, q_2)$ where $q_1 : Q \to X$ and $q_2 : Q \to Y$ are morphisms with $f q_1 = g q_2$, there must exist $u : Q \to P$ such that $p_1 \circ u = q_1$ and $p_2 \circ u = q_2$.

  The situation is illustrated in the following commutative diagram:
  \[
  \begin{tikzcd}
Q \arrow[rdd, "q_1"', bend right] \arrow[rrd, "q_2", bend left] \arrow[rd, "u", dashed] &                                      &                  \\
                                                                                        & P \arrow[r, "p_2"] \arrow[d, "p_1"'] & Y \arrow[d, "g"] \\
                                                                                        & X \arrow[r, "f"']                    & Z               
\end{tikzcd}
  \]
  This condition can be reformulated with universal arrows as follows: a pullback of $f$ and $g$ is a universal arrow from the functor $\Delta : \mathcal{C} \to \mathcal{C} \times \mathcal{C}$ to the object $(f, g)$ in $\mathcal{C} \times \mathcal{C}$.
\end{definition}

\begin{example}[Pullback in the category of categories]
  \label{ex:pullback_cat_cat}
  Let $\mathcal A, \mathcal B, \mathcal Z \in \Cat$ such that we have functors $F : \mathcal A \to \mathcal Z$ and $G : \mathcal B \to \mathcal Z$. 
  
  \medskip

  The pullback of $F$ and $G$ is the category $\mathcal P$ defined as follows:
\begin{itemize}
  \item The objects of $\mathcal P$ are pairs $(a, b)$ where $a$ is an object of $\mathcal A$, $b$ is an object of $\mathcal B$, and $Fa = Gb$.
  \item The morphisms of $\mathcal P$ from $(a, b)$ to $(a', b')$ are pairs $(f, g)$ where $f : a \to a'$ is a morphism in $\mathcal A$, $g : b \to b'$ is a morphism in $\mathcal B$, and $Ff = Gg$.
  \item The composition of morphisms in $\mathcal P$ is defined componentwise: if $(f, g) : (a, b) \to (a', b')$ and $(f', g') : (a', b') \to (a'', b'')$ are morphisms in $\mathcal P$, then their composition is given by $(f' \circ f, g' \circ g) : (a, b) \to (a'', b'')$.
  \item The identity morphism on an object $(a, b)$ in $\mathcal P$ is given by $(\text{id}_a, \text{id}_b)$, where $\text{id}_a$ is the identity morphism on $a$ in $\mathcal A$ and $\text{id}_b$ is the identity morphism on $b$ in $\mathcal B$.
\end{itemize}
\begin{proof}
    Consider the following commuting diagram:
    \[
    \begin{tikzcd}
      {\mathcal Q} && \\
      & {\mathcal P} & {\mathcal B} \\
      & {\mathcal A} & {\mathcal Z}
      \arrow["U"{description}, dashed, from=1-1, to=2-2]
      \arrow["{Q_2}", curve={height=-12pt}, from=1-1, to=2-3]
      \arrow["{Q_1}"', curve={height=12pt}, from=1-1, to=3-2]
      \arrow["{\pi_2}", from=2-2, to=2-3]
      \arrow["{\pi_1}"', from=2-2, to=3-2]
      \arrow["G", from=2-3, to=3-3]
      \arrow["F"', from=3-2, to=3-3]
    \end{tikzcd}
    \]
    We need to find $U$ and to prove that it is unique up to isomorphism.
    
    \medskip

    We define $U$ as follows: for any object $q$ in $\mathcal Q$, we have $Q_1 q \in \mathcal A$ and $Q_2 q \in \mathcal B$. Since $F Q_1 q = G Q_2 q$, we can define $U q = (Q_1 q, Q_2 q)$. For any morphism $m : q \to q'$ in $\mathcal Q$, we have $Q_1 m : Q_1 q \to Q_1 q'$ and $Q_2 m : Q_2 q \to Q_2 q'$. Since $F Q_1 m = G Q_2 m$, we can define $U m = (Q_1 m, Q_2 m)$.

    \medskip

    It is straightforward to check that $U$ is a functor and that the diagram commutes.
    
    \medskip
    
    Moreover, if $U' : \mathcal Q \to \mathcal P$ is another functor such that the diagram commutes, then for any object $q$ in $\mathcal Q$, we have $U' q = (Q_1 q, Q_2 q) = U q$. For any morphism $m : q \to q'$ in $\mathcal Q$, we have $U' m = (Q_1 m, Q_2 m) = U m$. Hence, $U$ is unique up to isomorphism.
\end{proof}
\end{example}

\begin{definition}[Pushout of two morphisms]
  Let $f : Z \to X$ and $g : Z \to Y$ be two morphisms in a category $\mathcal{C}$. 
  
  \medskip

  A \emph{pushout} of $f$ and $g$ consists of an object $P$ and two morphisms $p_1 : X \to P$ and $p_2 : Y \to P$ such that the following diagram commutes:
  \[
  \begin{tikzcd}
    Z \arrow[r, "g"] \arrow[d, "f"'] & Y \arrow[d, "p_2"]\\
    X \arrow[r, "p_1"'] & P
  \end{tikzcd}
  \]
  Moreover, for any triple $(Q, q_1, q_2)$ where $q_1 : X \to Q$ and $q_2 : Y \to Q$ are morphisms with $q_1 f = q_2 g$, there must exist $u : P \to Q$ such that $u \circ p_1 = q_1$ and $u \circ p_2 = q_2$.

  The situation is illustrated in the following commutative diagram:
  \[\begin{tikzcd}
    Z & Y & \\
    X & P \\
    && Q
    \arrow["g", from=1-1, to=1-2]
    \arrow["f"', from=1-1, to=2-1]
    \arrow["{p_2}", from=1-2, to=2-2]
    \arrow["{q_2}", curve={height=-12pt}, from=1-2, to=3-3]
    \arrow["{p_1}"', from=2-1, to=2-2]
    \arrow["{q_1}"', curve={height=12pt}, from=2-1, to=3-3]
    \arrow["u"{description}, dashed, from=2-2, to=3-3]
  \end{tikzcd}\]

  This condition can be reformulated with universal arrows as follows: a pushout of $f$ and $g$ is a universal arrow from the object $(f, g)$ in $\mathcal{C} \times \mathcal{C}$ to the functor $\Delta : \mathcal{C} \to \mathcal{C} \times \mathcal{C}$.
\end{definition}

\begin{definition}[Coequalizer]
  Let $f, g : X \to Y$ be two morphisms in a category $\mathcal{C}$. 
  
  \medskip
  
  A \emph{coequalizer} of $f$ and $g$ consists of an object $Q$ and a morphism $q : Y \to Q$ such that the following diagram commutes:
  \[
  \begin{tikzcd}
    X \arrow[r, shift left, "f"] \arrow[r, shift right, "g"'] & Y \arrow[r, "q"] & Q
  \end{tikzcd}
  \]
  Moreover, for any pair $(R, r)$ where $r : Y \to R$ is a morphism with $r \circ f = r \circ g$, there must exist a unique morphism $u : Q \to R$ such that $u \circ q = r$.

  The situation is illustrated in the following commutative diagram:
  \[
  \begin{tikzcd}
    X \arrow[r, shift left, "f"] \arrow[r, shift right, "g"'] & Y \arrow[r, "q"] \arrow[dr, "r"'] & Q \arrow[dashed, d, "u"]\\
    && R
  \end{tikzcd}
  \]
\end{definition}

\begin{lemma}[Pushout as coproducts and coequalizers]
  Let $f : Z \to X$ and $g : Z \to Y$ be two morphisms in a category $\mathcal{C}$. The pushout of $f$ and $g$ can be constructed as the coequalizer of the morphisms $\iota_X f : Z \to X \coprod Y$ and $\iota_Y g : Z \to X \coprod Y$, where $X \coprod Y$ is the coproduct of $X$ and $Y$, and $\iota_X : X \to X + Y$ and $\iota_Y : Y \to X \coprod Y$ are the canonical injections:
  \[\begin{tikzcd}
    Z && Y \\
    & {X\coprod Y} \\
    X && P
    \arrow["g", from=1-1, to=1-3]
    \arrow["{\tilde f}"', shift right, from=1-1, to=2-2]
    \arrow["{\tilde g}", shift left, from=1-1, to=2-2]
    \arrow["f"', from=1-1, to=3-1]
    \arrow["{\iota_Y}"{description}, from=1-3, to=2-2]
    \arrow["{\iota_Y\circ q}", from=1-3, to=3-3]
    \arrow["q"{description}, from=2-2, to=3-3]
    \arrow["{\iota_X}"{description}, from=3-1, to=2-2]
    \arrow["{\iota_X \circ q}"', from=3-1, to=3-3]
  \end{tikzcd}\]
\end{lemma}

\subsection{Monads}

\begin{definition}[Monad]
  A \emph{monad} on a category $\mathcal{C}$ is a triple $(T, \eta, \mu)$ where:
  \begin{itemize}
    \item $T: \mathcal{C} \to \mathcal{C}$ is a functor,
    \item $\eta: \text{Id}_{\mathcal{C}} \to T$ is a natural transformation called the unit,
    \item $\mu: T^2 \to T$ is a natural transformation called the multiplication,
  \end{itemize}
  such that the following diagrams commute:
  \[
  \begin{tikzcd}
    T \arrow[r, "T\eta"] \arrow[d, "\eta T"'] \arrow[equal, dr, ""] & T^2 \arrow[d, "\mu"] \\
    T^2 \arrow[r, "\mu"'] & T
  \end{tikzcd}
  \quad\quad
  \begin{tikzcd}
    T^3 \arrow[r, "T\mu"] \arrow[d, "\mu T"'] & T^2 \arrow[d, "\mu"] \\
    T^2 \arrow[r, "\mu"'] & T
  \end{tikzcd}
  \]
\end{definition}

\begin{example}[Monad from an Adjunction]
  Given an adjunction $F \dashv G$ between categories $\mathcal{C}$ and $\mathcal{D}$, we can construct a monad on $\mathcal{C}$ as follows:
  \begin{itemize}
    \item The functor $T: \mathcal{C} \to \mathcal{C}$ is defined by $T = GF$.
    \item The unit $\eta: \text{Id}_{\mathcal{C}} \to T$ is given by the natural transformation that assigns to each object $a$ in $\mathcal{C}$ the morphism $\eta_a: a \to GFa$ corresponding to the identity morphism on $Fa$ under the adjunction.
    \item The multiplication $\mu: T^2 \to T$ is given by the natural transformation that assigns to each object $a$ in $\mathcal{C}$ the morphism $\mu_a: GFGFa \to GFa$ corresponding to the composition of morphisms in $\mathcal{D}$ under the adjunction.
  \end{itemize}
\end{example}

\begin{definition}[Distributive law between monads]
  Given two monads $(T, \eta^T, \mu^T)$ and $(S, \eta^S, \mu^S)$ on a category $\mathcal{C}$, a \emph{distributive law} of $T$ over $S$ is a natural transformation $\lambda: TS \to ST$ such that the following diagrams commute:
  \[
  \begin{tikzcd}
                          & T \arrow[ld, "T\eta^S"'] \arrow[rd, "\eta^ST"] &    \\
TS \arrow[rr, "\lambda"'] &                                                & ST
\end{tikzcd}
  \quad\quad
  \begin{tikzcd}
                          & S \arrow[ld, "\eta^TS"'] \arrow[rd, "S\eta^T"] &    \\
TS \arrow[rr, "\lambda"'] &                                                & ST
\end{tikzcd}
  \]

  \[
  \begin{tikzcd}
    T^2 S \arrow[r, "T\lambda"] \arrow[d, "\mu^T S"'] & TST \arrow[r, "\lambda T"] & ST^2 \arrow[d, "S\mu^T"] \\
    TS \arrow[rr, "\lambda"'] & & ST
  \end{tikzcd}
  \quad\quad
  \begin{tikzcd}
    TS^2 \arrow[r, "\lambda S"] \arrow[d, "T\mu^S"'] & STS \arrow[r, "S\lambda"] & S^2T \arrow[d, "\mu^S T"] \\
    TS \arrow[rr, "\lambda"'] & & ST
  \end{tikzcd}
  \]

  This law induces a composite monad $ST$ on $\mathcal{C}$ with
  \begin{itemize}
    \item unit $\eta^{ST} = \eta^S \circ \eta^T$
    \item multiplication $STST \xrightarrow{S \lambda T} SSTT \xrightarrow {\mu^S \mu^T} ST$
  \end{itemize}
\end{definition}


\begin{definition}[Algebra over a monad]
  Let $(T, \eta, \mu)$ be a monad on a category $\mathcal{C}$. 
  
  \medskip
  
  An \emph{algebra over the monad $T$} (or simply a $T$-algebra) consists of:
  \begin{itemize}
    \item An object $A$ in $\mathcal{C}$,
    \item A morphism $a: TA \to A$ in $\mathcal{C}$, called the structure map, such that the following diagrams commute:
    \[
    \begin{tikzcd}
      A \arrow[r, "\eta_A"] \arrow[dr, "\text{id}_A"'] & TA \arrow[d, "a"] \\
      & A
    \end{tikzcd}
    \quad\quad
    \begin{tikzcd}
      T^2A \arrow[r, "Ta"] \arrow[d, "\mu_A"'] & TA \arrow[d, "a"] \\
      TA \arrow[r, "a"'] & A
    \end{tikzcd}
    \]
    The left one is called the unit property, and the right one is called the action property.
  \end{itemize}
\end{definition}


We'll now reformulate the definition of monads in terms of Kleisli categories and extension operators.

\begin{definition}[Kleisli category]
  Given a monad $(T, \eta, \mu)$ on a category $\mathcal{C}$, the \emph{Kleisli category} $\mathcal{C}_T$ has:
  \begin{itemize}
    \item The same objects as $\mathcal{C}$,
    \item For each pair of objects $a, b$, the morphisms from $a$ to $b$ in $\mathcal{C}_T$ are given by $\text{Hom}_{\mathcal{C}_T}(a, b) = \text{Hom}_{\mathcal{C}}(a, Tb)$,
    \item The identity morphism on an object $a$ is given by $\eta_a: a \to Ta$,
    \item The composition of morphisms $f: a \to Tb$ and $g: b \to Tc$ in $\mathcal{C}_T$ is given by the morphism $g \circ_T f: a \to Tc$ defined as the composition:
      \[
      a \xrightarrow{f} Tb \xrightarrow{Tg} T^2c \xrightarrow{\mu_c} Tc
      \]
  \end{itemize}
\end{definition}


\begin{definition}[Extension operator]
  Given a monad $(T, \eta, \mu)$ on a category $\mathcal{C}$, the \emph{extension operator} $(-)^\sharp$ is defined for each morphism $f: a \to Tb$ in $\mathcal{C}$ as the morphism $f^\sharp: Ta \to Tb$ given by the composition:
  \[
  Ta \xrightarrow{Tf} T^2b \xrightarrow{\mu_b} Tb
  \]
  In other words, $f^\sharp = \mu_b \circ Tf$.

  \medskip

  We have the following properties of the extension operator:
  \begin{itemize}
    \item $\eta_a^\sharp = \text{id}_{Ta}$ for all objects $a$ in $\mathcal{C}$.
    \item For any morphism $f: a \to Tb$, we have $f^\sharp \circ \eta_a = f$.
    \item For any morphisms $f: a \to Tb$ and $g: b \to Tc$, we have $(g^\sharp \circ f)^\sharp = g^\sharp \circ f^\sharp$.
  \end{itemize}
\end{definition}

\begin{definition}[Kleisli triple]
  A \emph{Kleisli triple} on a category $\mathcal{C}$ consists of:
  \begin{itemize}
    \item A mapping $T: \text{Ob}(\mathcal{C}) \to \text{Ob}(\mathcal{C})$,
    \item For each object $a$ in $\mathcal{C}$, a morphism $\eta_a: a \to Ta$,
    \item For each morphism $f: a \to Tb$ in $\mathcal{C}$, a morphism $f^\sharp: Ta \to Tb$,
  \end{itemize}
  such that the following conditions hold:
  \begin{itemize}
    \item $\eta_a^\sharp = \text{id}_{Ta}$ for all objects $a$ in $\mathcal{C}$.
    \item the following diagrams commute
    \[
    \begin{tikzcd}
      a \arrow[dr, "f"'] \arrow[r, "\eta_a"] & T a \arrow[d, "f^\sharp"] \\
      & T b
    \end{tikzcd}
    \quad\quad
    \begin{tikzcd}
      Ta \arrow[dr, "\lp g^\sharp \cdot f \rp^\sharp"'] \arrow[r, "f^\sharp"] & T b \arrow[d, "g^\sharp"] \\
      & T c
    \end{tikzcd}
    \]
  \end{itemize}
\end{definition}

\begin{proposition}
  Given a category $\mathcal{C}$, there is a one-to-one correspondence between monads on $\mathcal{C}$ and Kleisli triples on $\mathcal{C}$.
\end{proposition}

\begin{proof}
  Given a monad $(T, \eta, \mu)$ on $\mathcal{C}$, we can construct a Kleisli triple by defining the extension operator as $f^\sharp = \mu_b \circ Tf$ for each morphism $f: a \to Tb$.

  \medskip

  Conversely, given a Kleisli triple $(T, \eta, (-)^\sharp)$ on $\mathcal{C}$, we can construct a monad by defining the multiplication $\mu_a: T^2a \to Ta$ as $\mu_a = (\text{id}_{Ta})^\sharp$ for each object $a$ in $\mathcal{C}$.

  The conditions of the Kleisli triple ensure that the monad axioms are satisfied, and the construction is reversible, establishing the one-to-one correspondence.
\end{proof}

\begin{remarque}
  Kleisli triples are sometimes referred to as \emph{monads in extensive form}.
\end{remarque}

\begin{example}[Monads in terms of binds and returns]
  In the context of programming languages, a monad can be described in terms of a type constructor $T$ and two operations: \texttt{return} (or \texttt{unit}) and \texttt{bind}. The \texttt{return} operation corresponds to the unit $\eta$ of the monad, while the \texttt{bind} operation corresponds to the extension operator $(-)^\sharp$. Specifically, for a monad $(T, \eta, \mu)$ on a category $\mathcal{C}$, we can define:
  \begin{itemize}
    \item $\texttt{return}_a = \eta_a: a \to Ta$ for each object $a$ in $\mathcal{C}$,
    \item $\texttt{bind}_{a,b}(f) = f^\sharp: Ta \to Tb$ for each morphism $f: a \to Tb$ in $\mathcal{C}$.
  \end{itemize}
\end{example}

We use the notion of $\mathbb S$-algebra of \cite{Marmolejo2010MonadsAE} to define the distributive laws for monads in extensive form.

\begin{definition}[$\mathbb S$-algebra, \cite{Marmolejo2010MonadsAE}]
  Given a monad in extensive form $\mathbb S = (S, \eta, (-)^{\mathbb{S}})$ on a category $\mathcal C$, an \textit{$\mathbb S$-algebra} is a pair $\mathbb B = (B, (-)^{\mathbb B})$ where:
  \begin{itemize}
    \item $B \in \mathcal{C}$
    \item $(-)^{\mathbb B}_{X} : \mathcal C (X, B) \to \mathcal C(SX, B)$
  \end{itemize}
  such that the following diagrams commute for every $h : X \to B$ and $k : Y \to SX$:
  \[
  \begin{tikzcd}
    X \arrow[r, "\eta X"] \arrow[dr, "h"'] & SX \arrow[d, "{h^{\mathbb B}}"]\\
    & B
  \end{tikzcd}
  \quad\quad
  \begin{tikzcd}
    SY \arrow[r, "{k^{\mathbb S}}"] \arrow[dr, "{(h^{\mathbb B} k)^{\mathbb B}}"'] & SX \arrow[d, "{h^{\mathbb B}}"] \\
    & B
  \end{tikzcd}
  \]
  A morphism of $\mathbb S$-algebras $(B, (-)^\mathbb{B})$ to $(A, (-)^\mathbb{A})$ can be defined as an arrow $l : B \to A$ in $\mathcal C$ such that the following diagram commutes for every $h : X \to B$:
  \[
  \begin{tikzcd}
    SX \arrow[r, "h^\mathbb{B}"] \arrow[dr, "{(l \cdot h)^\mathbb{A}}"'] & B \arrow[d, "l"] \\
    & A
  \end{tikzcd}
  \]
\end{definition}


\begin{theorem}[Distributive law between monads in extensive form, \cite{Marmolejo2010MonadsAE}]
Let $\mathbb T = (T, \eta_T, (-)^{\mathbb T})$ and $\mathbb S = (S, \eta_S, (-)^{\mathbb S})$ be monads on $\mathcal C$. A distributive law of $\mathbb T$ over $\mathbb S$ can equivalently be given by an $\mathbb T$-algebra $(STA, (-)^\lambda)$ for every $A \in \mathcal C$ such that
\begin{itemize}
  \item $(S\eta_TA \cdot \eta_SA)^\lambda = \eta_S TA$
  \item for every $f : B \to TSA$, $(f^\lambda)^\mathbb{S} : (STB, (-)^\lambda) \to (STA, (-)^\lambda)$ is a morphism of $\mathbb T$-algebras.
\end{itemize}
Unfolding the conditions for the $\mathbb T$-algebra and for the morphism, the following diagrams must commute for every $h : X \to STA$, $k : Y \to TX$, $f : B \to STA$ and $g : X \to STB$:
\[
\begin{tikzcd}
    X \arrow[r, "\eta_T X"] \arrow[dr, "h"'] & TX \arrow[d, "{h^{\lambda}}"]\\
    & STA
  \end{tikzcd}
  \quad\quad
  \begin{tikzcd}
    TY \arrow[r, "{k^{\mathbb T}}"] \arrow[dr, "{(h^{\lambda} k)^{\lambda}}"'] & TX \arrow[d, "{h^{\lambda}}"] \\
    & STA
  \end{tikzcd}
  \quad\quad
  \begin{tikzcd}
    TB \arrow[r, "g^\lambda"] \arrow[dr, "{\lp (f^\lambda)^\mathbb{S} \cdot g\rp^\lambda}"'] & STB \arrow[d, "(f^\lambda)^\mathbb{S}"] \\
    & STA
  \end{tikzcd}
\]
\end{theorem}

\begin{proposition}[\cite{Marmolejo2010MonadsAE}]
  Given a distributive law $\lambda$ of $\mathbb T$ over $\mathbb S$, the composite monad is given by $\mathbb S \circ_\lambda \mathbb T = (ST, S\eta_T \cdot \eta_S, \lp(-)^\lambda)^\mathbb{S}\rp$
\end{proposition}

\begin{proposition}[\cite{Marmolejo2010MonadsAE}]
  Let $\mathbb S$ and $\mathbb T$ be monads on $\mathcal C$. There is a bijection between distributive laws of $\mathbb T$ over $\mathbb S$ and monad structures $(ST, S\eta_T \cdot \eta_S, (-)^{(\mathbb{ST})})$ on $ST$ such that:
  \begin{itemize}
    \item for every $k : Y \to TX, S(k^\mathbb{T}) = (\eta_S TX \cdot k)^{(\mathbb{ST})}$
    \item for every $m : Y \to ST X$ and $h : X \to STU$:
    \begin{itemize}
      \item $h^{(\mathbb{ST})} = (h^{(\mathbb{ST})} \cdot \eta_S TX)^\mathbb{T}$
      \item the following diagram commutes
      \[
      \begin{tikzcd}
        SY \arrow[r, "m^\mathbb{S}"] \arrow[dr, "\lp h^{(\mathbb{ST})} \cdot m \rp^\mathbb{S}"'] & STX \arrow[d, "h^{(\mathbb{ST})}"] \\
        & STU
      \end{tikzcd}
      \]
    \end{itemize}
  \end{itemize}
  \end{proposition}

\subsection{2-categories}

\begin{definition}[2-category]
  A \emph{2-category} $C$ consists of:
  \begin{itemize}
    \item Cellular structure:
    \begin{itemize}
      \item A collection $C_0$ of 0-cells $a, b, c, \dots$ (objects),
      \item For each pair of 0-cells $a, b$, a collection $C_1(a, b)$ of 1-cells (morphisms) from $a$ to $b$,
      \item For each pair of 1-cells $f, g \in C_1(a, b)$, a collection $C_2(f, g)$ of 2-cells (2-morphisms) from $f$ to $g$
    \end{itemize}
    \item Identities:
    \begin{itemize}
      \item A 1-cell $\id_a : a \to a$ for each 0-cell $a \in C_0$ (identity 1-cell),
      \item A 2-cell $\id_f : f \imp f$ for each 1-cell $f \in C_1(a, b)$ (identity 2-cell)
    \end{itemize}
    \item Composition:
    \begin{itemize}
      \item A composition operation on 1-cells $g \circ f$ for every pair of composable 1-cells $f: a \to b$ and $g: b \to c$,
      \item A vertical composition operation on 2-cells $\beta \circ \alpha : f \imp h$ for every pair of composable 2-cells $\alpha: f \imp g$ and $\beta: g \imp h$,
      \item A horizontal composition operation on 2-cells $\beta \bullet \alpha : g \circ f \imp g' \circ f'$ for every pair of composable 2-cells $\alpha: f \imp f'$ and $\beta: g \imp g'$
    \end{itemize}
    such that
    \begin{itemize}
      \item 1-cell compositions are associative and identities are neutral
      \item 2-cell vertical compositions are associative and identities are neutral
      \item 2-cell horizontal compositions are associative and identities are neutral, and composition of 2-cell identities preserves 1-cell composition
      \item the interchange law holds: for any composable 2-cells $\alpha, \alpha', \beta, \beta'$, we have $$(\beta' \circ \beta) \bullet (\alpha' \circ \alpha) = (\beta' \bullet \alpha') \circ (\beta \bullet \alpha)$$
    \end{itemize}
  \end{itemize}
\end{definition}

\begin{example}[2-category of Categories]
  The 2-category $\Cat$ has:
  \begin{itemize}
    \item 0-cells: categories,
    \item 1-cells: functors between categories,
    \item 2-cells: natural transformations between functors.
    \item pointwise composition of functors as 1-cell composition,
    \item pointwise composition of natural transformations as vertical composition,
    \item horizontal composition of natural transformations defined by the composition of functors.
  \end{itemize}
\end{example}

\begin{definition}[Adjunction in a 2-category]
  Given a 2-category $C$, an \emph{adjunction} between 0-cells $a$ and $b$ in $C$ consists of:
  \begin{itemize}
    \item Two 1-cells $f: a \to b$ and $g: b \to a$,
    \item Two 2-cells $\eta: \id_a \imp g \circ f$ (unit) and $\epsilon: f \circ g \imp \id_b$ (counit),
  \end{itemize}
  such that the following triangle identities hold:
  \[
  (f \bullet \eta) \circ (\epsilon \bullet f) = \id_f
  \quad\quad
  (\eta \bullet g) \circ (g \bullet \epsilon) = \id_g
  \]
\end{definition}

\begin{definition}[2-functor]
  Given two 2-categories $C$ and $D$, a \emph{2-functor} $F: C \to D$ consists of three mappings:
  \begin{itemize}
    \item $F: C_0 \to D_0$ that assigns to each 0-cell $a$ in $C$ a 0-cell $Fa$ in $D$,
    \item $F: C_1(a, b) \to D_1(Fa, Fb)$ for each pair of 0-cells $a, b$ in $C$, that assigns to each 1-cell $f: a \to b$ in $C$ a 1-cell $Ff: Fa \to Fb$ in $D$,
    \item $F: C_2(f, g) \to D_2(Ff, Fg)$ for each pair of 1-cells $f, g: a \to b$ in $C$, that assigns to each 2-cell $\alpha: f \imp g$ in $C$ a 2-cell $F\alpha: Ff \imp Fg$ in $D$,
  \end{itemize}
  such that $F$ preserves identities and compositions of 1-cells and 2-cells.
\end{definition}

\begin{lemma}
  Let $F : C \to D$ be a 2-functor.

  \medskip

  If $\la f : a \to b, u, \eta, \epsilon \ra$ is an adjunction in $C$, then $\la Ff : Fa \to Fb, Fu, F\eta, F\epsilon \ra$ is an adjunction in $D$.
\end{lemma}

\begin{proof}
  The candidate adjunction has the right type, and a 2-functor preserves pasting diagrams and identities, so the triangle identities are preserved.
\end{proof}


\begin{definition}[natural 2-transformation]
  Let $F, G : C \to D$ be two 2-functors between 2-categories.

  \medskip

  A \emph{(strict) natural 2-transformation} \begin{tikzcd}
	C & D
	\arrow[""{name=0, anchor=center, inner sep=0}, "G"', curve={height=12pt}, from=1-1, to=1-2]
	\arrow[""{name=1, anchor=center, inner sep=0}, "F", curve={height=-12pt}, from=1-1, to=1-2]
	\arrow["\Xi", between={0.2}{0.8}, Rightarrow, from=1, to=0]
\end{tikzcd} consists of a 1-cell $\Xi_a : F_0 a \to G_0 a$ for each 0-cell $a$ such that:
\begin{itemize}
  \item for every 1-cell $f : a \to b$ in $C$: \[\begin{tikzcd}
	{F_0a} & {F_0b} \\
	{G_0a} & {G_0b}
	\arrow["{F_1f}", from=1-1, to=1-2]
	\arrow["{\Xi_a}"', from=1-1, to=2-1]
	\arrow["{\Xi_b}", from=1-2, to=2-2]
	\arrow["{G_1f}"', from=2-1, to=2-2]
\end{tikzcd}\]
\item for every 2-cell \begin{tikzcd}
	a & b
	\arrow[""{name=0, anchor=center, inner sep=0}, "f", curve={height=-6pt}, from=1-1, to=1-2]
	\arrow[""{name=1, anchor=center, inner sep=0}, "g"', curve={height=6pt}, from=1-1, to=1-2]
	\arrow["\alpha", between={0.2}{0.8}, Rightarrow, from=0, to=1]
\end{tikzcd}:
\[\begin{tikzcd}
	& Ga &&&& Ga & \\
	Fa && Gb & {=} & Fa && Gb \\
	& Fb &&&& Fb
	\arrow[""{name=0, anchor=center, inner sep=0}, "Gf", curve={height=-12pt}, from=1-2, to=2-3]
	\arrow[""{name=1, anchor=center, inner sep=0}, "Gg"', curve={height=12pt}, from=1-2, to=2-3]
	\arrow["Gf", from=1-6, to=2-7]
	\arrow["{\Xi_a}", from=2-1, to=1-2]
	\arrow["Fg"', from=2-1, to=3-2]
	\arrow["{\Xi_a}", from=2-5, to=1-6]
	\arrow[""{name=2, anchor=center, inner sep=0}, "Fg"', curve={height=12pt}, from=2-5, to=3-6]
	\arrow[""{name=3, anchor=center, inner sep=0}, "Ff", curve={height=-12pt}, from=2-5, to=3-6]
	\arrow["{\Xi_b}"', from=3-2, to=2-3]
	\arrow["{\Xi_b}"', from=3-6, to=2-7]
	\arrow["{G\alpha}", between={0.2}{0.8}, Rightarrow, from=0, to=1]
	\arrow["{F\alpha}", between={0.2}{0.8}, Rightarrow, from=3, to=2]
\end{tikzcd}\]
\end{itemize}
\end{definition}

\subsection{Multicategories}

\begin{definition}[Multicategory]
  A multicategory $\mathcal{U}$ consists of:
  \begin{itemize}
    \item A class of objects $\text{Ob}(\mathcal{U})$,
    \item For each finite (possibly empty) sequence of objects $(a_1, a_2, ..., a_n)$ and an object $b$, a set of morphisms ${\mathcal{U}}(a_1, a_2, ..., a_n; b)$,
    \item For each object $a$, an identity morphism $\text{id}_a : (a) \to a$,
    \item A composition operation that allows for the composition of morphisms in a way that generalizes the composition in a category, satisfying appropriate associativity and identity axioms.
  \end{itemize}
\end{definition}

\begin{example}[Category with Finite Products as a Multicategory]
  Given a category $\mathcal{C}$ with finite products, we can construct a multicategory $\mathcal{U}$ where:
  \begin{itemize}
    \item The objects of $\mathcal{U}$ are the same as the objects of $\mathcal{C}$,
    \item The morphisms ${\mathcal{U}}(a_1, a_2, ..., a_n; b)$ are given by the morphisms in $\mathcal{C}$ from the product $a_1 \times a_2 \times ... \times a_n$ to $b$,
    \item The identities rely on the canonical isomorphism $$\text{id}_a := \prodd{i=1
    }na \cong a$$
    \item The composition is defined using the universal property of products and the composition in $\mathcal{C}$.
  \end{itemize}
  \label{ex:cat_product_multicat}
\end{example}

\begin{example}[Monoidal Category as a Multicategory]
  A monoidal category $(\mathcal{C}, \otimes, I)$ can be viewed as a multicategory where:
  \begin{itemize}
    \item The objects of the multicategory are the same as the objects of $\mathcal{C}$,
    \item The morphisms ${\mathcal{U}}(a_1, a_2, ..., a_n; b)$ are given by the morphisms in $\mathcal{C}$ from $a_1 \otimes a_2 \otimes ... \otimes a_n$ to $b$ (by nesting the monoidal product)
  \end{itemize}
\end{example}

\begin{definition}[Multifunctor]
  A \emph{multi-functor} $F: \mathcal{U} \to \mathcal{V}$ between multicategories $\mathcal{U}$ and $\mathcal{V}$ consists of two mappings:
  \begin{itemize}
    \item A mapping $F: \text{Ob}(\mathcal{U}) \to \text{Ob}(\mathcal{V})$ that assigns to each object $a$ in $\mathcal{U}$ an object $Fa$ in $\mathcal{V}$,
    \item A mapping $F: {\mathcal{U}}(a_1, a_2, ..., a_n; b) \to {\mathcal{V}}(Fa_1, Fa_2, ..., Fa_n; Fb)$ that assigns to each morphism $f: (a_1, a_2, ..., a_n) \to b$ in $\mathcal{U}$ a morphism $Ff: (Fa_1, Fa_2, ..., Fa_n) \to Fb$ in $\mathcal{V}$,
  \end{itemize}
  such that $F$ preserves:
  \begin{itemize}
    \item Identities: For each object $a$ in $\mathcal{U}$, $F\text{id}_a = \text{id}_{Fa}$,
    \item Composition: For morphisms $f: (a_1, a_2, ..., a_n) \to b$ and $g_i: (a_{i1}, a_{i2}, ..., a_{ik_i}) \to a_i$ in $\mathcal{U}$, we have
      \[
      F(f \circ (g_1, g_2, ..., g_n)) = Ff \circ (Fg_1, Fg_2, ..., Fg_n).
      \]
  \end{itemize}
\end{definition}


\begin{definition}[Multi-natural transformation]
  Given two multi-functors $F, G: \mathcal{U} \to \mathcal{V}$ between multicategories $\mathcal{U}$ and $\mathcal{V}$, a \emph{multi-natural transformation} $\eta: F \to G$ consists of a family of morphisms $\{\eta_a: Fa \to Ga\}_{a \in \text{Ob}(\mathcal{U})}$ such that for every morphism $f: (a_1, a_2, ..., a_n) \to b$ in $\mathcal{U}$, the following diagram commutes:
  \[
  \begin{tikzcd}
    (Fa_1, Fa_2, ..., Fa_n) \arrow[r, "Ff"] \arrow[d, "{(\eta_{a_1}, \eta_{a_2}, ..., \eta_{a_n})}"'] & Fb \arrow[d, "\eta_b"] \\
    (Ga_1, Ga_2, ..., Ga_n) \arrow[r, "Gf"'] & Gb
  \end{tikzcd}
  \]
\end{definition}

\begin{example}[Category of multicategories]
  We have a 2-category $\Multicat$ where:
  \begin{itemize}
    \item 0-cells are multi-categories
    \item 1-cells are multi-functors
    \item 2-cells are multi-natural transformations
  \end{itemize} with multi-functor composition and identities: $$\Id_C a := a \quad \Id_C f := f$$ vertical, horizontal composition of multi-natural transformations, with identities: $$(\id_F)_a := \id_{Fa} : (a) \to a$$
\end{example}

\begin{example}[Ordinary 2-functor]
  We have a 2-functor structure $-_{o} : \Multicat \to \Cat$ sending:
  \begin{itemize}
    \item each multi-category $\mathcal U$ to its (\textbf{o}rdinary) category of morphisms:
    \begin{itemize}
      \item $\Ob(\mathcal U_o) := \Ob (\mathcal U)$
      \item $\mathcal U_o(a, b) := \mathcal U (a; b)$, i.e the morphisms in $\mathcal U_o$ are the multi-morphisms with a length-1 source
      \item identities and composition are given by identity morphisms and composition in the multi-category
    \end{itemize}
    \item each multi-functor $F : \mathcal U \to \mathcal V$ to the functor sending:
    \begin{itemize}
      \item $a \in \Ob(\mathcal U)$ to $Fa \in \Ob(\mathcal V)$
      \item $f : (a) \to b$ to $F_of := Ff : (Fa) \to Fb$
    \end{itemize}
    \item each multi-natural transformation \begin{tikzcd}
	{\mathcal U} & {\mathcal V}
	\arrow[""{name=0, anchor=center, inner sep=0}, "G"', curve={height=12pt}, from=1-1, to=1-2]
	\arrow[""{name=1, anchor=center, inner sep=0}, "F", curve={height=-12pt}, from=1-1, to=1-2]
	\arrow["\alpha", between={0.2}{0.8}, Rightarrow, from=1, to=0]
\end{tikzcd} to $(\alpha_o)_a := \alpha_a$
  \end{itemize}
  It is 2-faithful.
\end{example}


\subsection{Operads}

\begin{definition}[Operad]
  An \emph{operad} $\mathcal{O}$ consists of:
  \begin{itemize}
    \item A collection of sets $\{\mathcal{O}_n\}_{n \geq 0}$, where $\mathcal{O}_n$ is the set of $n$-ary operations,
    \item A composition function $\gamma: \mathcal{O}_n \times \mathcal{O}_{k_1} \times \cdots \times \mathcal{O}_{k_n} \to \mathcal{O}_{k_1 + \cdots + k_n}$,
    \item An identity element $e \in \mathcal{O}_1$,
  \end{itemize}
  satisfying the following axioms:
  \begin{itemize}
    \item Associativity: For all $f \in \mathcal{O}_n$, $g_i \in \mathcal{O}_{k_i}$, and $h_{ij} \in \mathcal{O}_{m_{ij}}$, we have
      \[
      \gamma(f; g_1, ..., g_n) = f(g_1(h_{11}, ..., h_{1k_1}), ..., g_n(h_{n1}, ..., h_{nk_n})).
      \]
    \item Identity: For all $f \in \mathcal{O}_n$, we have
      \[
      f(e, ..., e) = f.
      \]
  \end{itemize}
\end{definition}

\begin{definition}[Algrebraic Theory]
  An \emph{algebraic theory} $\mathcal{T}$ consists of:
  \begin{itemize}
    \item A collection of sets $\{\mathcal{T}_n\}_{n \geq 0}$, where $\mathcal{T}_n$ is the set of $n$-ary operations,
    \item A collection of equations between terms built from these operations,
  \end{itemize}
  such that the operations and equations define a class of algebras (models) that satisfy the given equations.
\end{definition}

\begin{example}[Operad from an Algebraic Theory]
  Given an algebraic theory $\mathcal{T}$, we can construct an operad $\mathcal{O}^{\mathcal T}$ where:
  \begin{itemize}
    \item $\mathcal{O}_n^{\mathcal T}$ consists of the $n$-ary operations in $\mathcal{T}$,
    \item The composition function $\gamma$ is defined by substituting operations according to the equations of the theory,
    \item The identity element $e$ corresponds to the identity operation in the theory.
  \end{itemize}
\end{example}

\begin{definition}[Renaming]
  A \emph{renaming} $\rho : n \to m$ is any function from the set of $n$ variables to the set of $m$ variables.
\end{definition}

\begin{example}[of renaming]
  Let $\rho : 3 \to 2$ be the renaming defined by $\rho(x_1) = y_1$, $\rho(x_2) = y_2$, and $\rho(x_3) = y_1$. Then for any term $t \in \pres_3$, we have $\rho(t) \in \pres_2$ obtained by substituting $y_1$ for $x_1$ and $x_3$, and $y_2$ for $x_2$ in $t$.
\end{example}

\begin{definition}[Translation]
  Let $\mathcal O$ be an operad and $\pres$ be a theory.

  \medskip

  A \emph{translation} $\mathfrak T : \mathcal O \to \pres$ consists of an assignment $\mathfrak T f \in \pres_n$ for every $f \in \mathcal O_n$ such that:
  \begin{itemize}
    \item the identity $\id : \mathcal O_1$ is sent the the unique variable projection: $\mathfrak T \id = x \in \pres_1$
    \item the translation respects composition, for all $f \in \mathcal O_n$ composable with $g_i \in \mathcal O_{m_i}$: $$\mathfrak T (f \circ (g_1, \dots, g_n)) = \mathfrak T f [x_i \mapsto \mathfrak T g_i]_{i=1}^n$$
  \end{itemize}
  A translation is \emph{surjective}, written $\mathfrak T : \mathcal O \twoheadrightarrow \pres$, if for every term $t \in \pres_n$, there exists an operator $f \in \mathcal O_m$ and a renaming $\rho : m \to n$ such that $t = \mathfrak T f[\rho]$.
\end{definition}

\begin{definition}[Model]
  Let $\mathcal O$ be an operad, and $\mathcal U$ be a multicategory. A \emph{model} of $\mathcal O$ in $\mathcal U$ is a multi-functor $\mathcal A: \mathcal O \to \mathcal U$ that preserves the operadic structure, i.e., it respects the composition and identity of the operad.
\end{definition}

\begin{definition}{Homomorphism of Models}
  A \emph{homomorphism} $h : \mathcal A \to \mathcal B$ between $\mathcal O$-models in $\mathcal U$ consists of a morphism $\underline{h} : (\underline{\mathcal A}) \to \underline{\mathcal B}$ such that for all $f \in \mathcal O_n$, $$\underline{h} \circ (\mathcal A \brak{f}) = \mathcal B \brak{f} (\underline{h}, \dots, \underline{h})$$
\end{definition}

\begin{example}[Category of Models]
  Given an operad $\mathcal O$ and a multicategory $\mathcal U$, we can construct a category $\Mod(\mathcal O, \mathcal U)$ where:
  \begin{itemize}
    \item The objects are the models of $\mathcal O$ in $\mathcal U$,
    \item The morphisms are the homomorphisms between these models.
  \end{itemize}
\end{example}

\begin{proposition}
  \label{prop:mod_translation_functor}
  Let $\mathfrak T : \mathcal O \to \pres$ be a translation, and let $\mathcal C$ be a category with chosen finite product $\prod$.

  \medskip

  There is a carrier-preserving, fully-faithful functor $\Mod(\mathfrak T, \la \mathcal C, \prod \ra) : \Mod(\pres, \mathcal C) \to \Mod(\mathcal O, \la \mathcal C, \prod \ra)$ given by: $$\lp \Mod(\mathfrak T, \la \mathcal C, \prod \ra) A \rp \brak{f} := A \brak{\mathfrak T f} \qquad \Mod(\mathfrak T, \la \mathcal C, \prod \ra) h := h$$
\end{proposition}

\begin{proposition}
  The compatibility with composition and identities of the operad follows from their preservation by $\mathfrak T$. Preservation of identities and composition, and fully-faithfulness are evident.
\end{proposition}

\begin{proposition}
  \label{prop:mod_2_functor}
  We define a 2-functor $\Mod(\mathcal O, -) : \Multicat \to \Cat$ such that:
  \begin{itemize}
    \item for every multi-category $\mathcal U$, $\Mod(\mathcal O, \mathcal U)$ is the category of $\mathcal O$-models in $\mathcal U$.
    \item for every multi-functor $F : \mathcal U \to \mathcal V$, the functor $\Mod(\mathcal O, F) : \Mod(\mathcal O, \mathcal U) \to \Mod(\mathcal O, \mathcal V)$ sends:
    \begin{itemize}
      \item each $\mathcal O$-model $\mathcal A$ in $\mathcal U$ to the $\mathcal O$-model:
      \begin{itemize}
        \item $\underline{F \mathcal A} := F \underline{\mathcal A}$
        \item $(F \mathcal A) \brak{f} : (F \underline{\mathcal A}, \dots, F \underline{\mathcal A}) \xrightarrow{F(\mathcal A\brak f)} F \underline{\mathcal A}$
      \end{itemize}
      \item each $\mathcal O$-homomorphism $h : \mathcal A \to \mathcal B$ to the $\mathcal O$-homomorphism over $F \underline{h} : F \mathcal A \to F \mathcal B$
    \end{itemize}
    \item for every multi-natural transformation \begin{tikzcd}
	{\mathcal U} & {\mathcal V}
	\arrow[""{name=0, anchor=center, inner sep=0}, "F", curve={height=-12pt}, from=1-1, to=1-2]
	\arrow[""{name=1, anchor=center, inner sep=0}, "G"', curve={height=12pt}, from=1-1, to=1-2]
	\arrow["\alpha", between={0.2}{0.8}, Rightarrow, from=0, to=1]
\end{tikzcd} we have a natural transformation\\ \begin{tikzcd}
	{\Mod(\mathcal O, \mathcal U)} &&& {\Mod(\mathcal O, \mathcal V)}
	\arrow[""{name=0, anchor=center, inner sep=0}, "{\Mod(\mathcal O, F)}", curve={height=-12pt}, from=1-1, to=1-4]
	\arrow[""{name=1, anchor=center, inner sep=0}, "{\Mod(\mathcal O, G)}"', curve={height=12pt}, from=1-1, to=1-4]

  \end{itemize}
\end{proposition}



\subsection{Free algebras}

We'll use the definitions from \cite{Allais2025}.

\begin{definition}[Finitary signature]
  A \emph{finitary signature} $\Sigma = (\op \Sigma, \arity)$ consists of a set $\op \Sigma$ of \emph{operators}, and an assigment $\arity : \op \Sigma \to \N$ associating to each operator in $\op \Sigma$ its \emph{arity}.
\end{definition}

\begin{definition}[Algebra]
  An \emph{algebra} $A = (\underline{A}, A \brak{\text{--}})$ over a finitary signature $\Sigma$ consists of:
  \begin{itemize}
    \item A set $\underline{A}$ called the \emph{carrier} of $A$,
    \item For each operator $f \in \op \Sigma$ of arity $n$, a function $A \brak{f} : \underline{A}^n \to \underline{A}$ called the \emph{interpretation} of $f$ in $A$.
  \end{itemize}
\end{definition}

\begin{definition}[Terms]
  Given a set $X$ of variables and a finitary signature $\Sigma$, the set of \emph{terms} $\Term_\Sigma(X)$ is defined inductively as follows:
  \begin{itemize}
    \item Every variable $x \in X$ is a term,
    \item If $f \in \op \Sigma$ is an operator of arity $n$ and $t_1, t_2, ..., t_n$ are terms, then $f(t_1, t_2, ..., t_n)$ is a term.
  \end{itemize}
\end{definition}

\begin{remarque}
  The definition of algebraic theory seen before can be reformulated as a finitary signature together with a set of equations between terms built from the operators in the signature.
\end{remarque}

\begin{definition}[Environment and Interpretation]
  An \emph{environment} $\rho$ for a context $X$ in an algebra $A$ is a function $\rho : X \to \underline{A}$ that assigns to each variable in $X$ an element of the carrier of $A$.
  
  \medskip

  Given a term $t \in \Term_\Sigma(X)$, denoted $X \vdash t$, the algebra $A$ gives an \emph{interpretation} function $A \brak{t} : \underline{A}^X \to \underline{A}$. Given an environment $\rho : X \to \underline{A}$, we can evaluate the term $t$ in $A$ under the environment $\rho$, denoted $A \brak{t}(\rho)$, by recursively substituting the variables in $t$ with their corresponding values in $\underline{A}$ according to $\rho$ and applying the interpretations of the operators as defined in $A$.
\end{definition}

\begin{definition}[Valid equations]
  An equation $t = t'$ between terms in $\Term_\Sigma(X)$ is \emph{valid} in an algebra $A$ if for every environment $\rho : X \to \underline{A}$, we have $A \brak{t}(\rho) = A \brak{t'}(\rho)$.
\end{definition}


\begin{definition}[Presentation and Models]
  A \emph{presentation} $\mathcal T = (\Sigma_{\mathcal T}, \mathcal T_{ax})$ consists of a finitary signature $\Sigma_{\mathcal T}$ and a set $\mathcal T_{ax}$ of \emph{axioms}, which are equations between terms in $\Term_{\Sigma_{\mathcal T}}(X)$ for some set of variables $X$.

  \medskip

  A $\mathcal T$-model $A$ is a $\Sigma_{\mathcal T}$-algebra such that all equations in $\mathcal T_{ax}$ are valid in $A$.

  \medskip

  A $\mathcal T$-model $A$ over a context $X$ is a $\mathcal T$-model $A$ together with an environment $\rho : X \to \underline{A}$.
\end{definition}

\begin{definition}[Homomorphisms]
  Let $A = (\underline{A}, A \brak{\text{--}})$ and $B = (\underline{B}, B \brak{\text{--}})$ be algebras over the same signature $\Sigma$. 

  \medskip
  
  A \emph{homomorphism} of $\Sigma$-algebras from $A$ to $B$ is a function $h : \underline{A} \to \underline{B}$ such that for every operator $f \in \op \Sigma$ of arity $n$, we have:
  $$B \brak{f}(h(a_1), \ldots, h(a_n)) = h(A \brak{f}(a_1, \ldots, a_n))$$
  for all $a_1, \ldots, a_n \in \underline{A}$.

  \medskip

  A \emph{homomorphism} of $\mathcal T$-models is a homomorphism of $\Sigma_{\mathcal T}$-algebras between the underlying algebras of the models.
\end{definition}

\begin{definition}[Morphism]
  A \emph{morphism} $h : A \to B$ of $\mathcal T$-models over $X$ is a $\mathcal T$-model homomorphism that makes the diagram commute:
  \[  \begin{tikzcd}
    X \arrow[r, "\rho_A"] \arrow[dr, "\rho_B"'] & \underline{A} \arrow[d, "h"] \\
    & \underline{B}
  \end{tikzcd}
  \]
\end{definition}

\begin{definition}[Free model (or free algebra)]
  Let $\mathcal T$ be a presentation and $X$ be a set of variables. 

  \medskip
  
  A \emph{free $\mathcal T$-model} over $X$ is a $\mathcal T$-model $F(X)$ such that for any $\mathcal T$-model $A$ over $X$, there exists a unique morphism of $\mathcal T$-models from $F(X)$ to $A$.

  \medskip 

  The existence-and-uniqueness property is called the \emph{universal property} of free $\mathcal T$-models over $X$. 
\end{definition}

Informaly:
\begin{itemize}
  \item A signature gives defines the operators and constants (arity 0 operators).
  \item An algebra gives a concrete interpretation of the signature.
  \item A presentation defines axioms on top of the signature, for instance the axioms of a group, or of a semiring, etc.
  \item A model of a presentation is an algebra that satisfies the axioms of the presentation.
  \item A free algebra is the most general model of a presentation.
\end{itemize}

The \emph{free extension} of $\mathcal T$ by $X$, denoted $\mathcal T[X]$, is the presentation obtained by adding the variables in $X$ as new constants (arity 0 operators) to the signature of $\mathcal T$, and keeping the same axioms.

We can give an equivalent definition for free algebras using universal arrows.

\begin{proposition}
  Let $\pres$ be a presentation, and $X$ be a set. Let $\pres$-$\Mod$ be the category of $\pres$-models over $X$. Let $U : \pres\text{-}\Mod \to \Set$ be the forgetful functor that sends a $\pres$-model to its underlying set. 
  
  \medskip

  Then, a free $\pres$-model over $X$ is a universal arrow from $X$ to $U$.
\end{proposition}

\begin{proof}
  Let $F\,X$ be a free $\pres$-model over $X$. We want to show that $(F\,X, \eta_X)$ is a universal arrow from $X$ to $U$, where $\eta_X : X \to UF\,X$ is the function that sends each variable in $X$ to its interpretation in $F\,X$.

  \medskip

  Let $A$ be a $\pres$-model over $X$, and let $\rho_A : X \to U\,A$ be the environment of $A$. Since $F\,X$ is a free $\pres$-model over $X$, there exists a unique morphism of $\pres$-models $h : F\,X \to A$ such that the following diagram commutes:
  \[
  \begin{tikzcd}
    X \arrow[r, "\eta_X"] \arrow[dr, "\rho_A"'] & UF\,X \arrow[d, "U\,h"] \\
    & U\,A
  \end{tikzcd}
  \]
  
  This shows that $(F\,X, \eta_X)$ satisfies the universal property of being a universal arrow from $X$ to $U$. 
  
  \medskip
  
  Conversely, if $(F\,X, \eta_X)$ is a universal arrow from $X$ to $U$, then for any $\pres$-model $A$ over $X$, there exists a unique morphism of $\pres$-models from $F\,X$ to $A$, which means that $F\,X$ is a free $\pres$-model over $X$. Therefore, the two definitions are equivalent.
\end{proof}


\subsection{Multi-homomorphisms}

This section will be a subset of Ohad's note. In particular, all proofs will be omitted.

\medskip

In all of this section, we fix an algebraic theory $\mathcal T$ and $\la \mathcal C, \prod \ra$ a category with chosen finite products.

\begin{definition}[Strength]
  For all $k \in \N^*$ and $X, Y \in \mathcal C$, we define the \emph{strength} (for the reader monad $X^k$) by $$\str : X^k \times Y \xrightarrow{\la \pi_i \times \id \ra_{i=1}^k} (X \times Y)^k$$
\end{definition}

\begin{definition}[Multi-$\mathcal T$-homomorphism]
  Let $A_1, A_2, \dots, A_k$ and $B$ be $\mathcal T$-models in $\mathcal C$. 
  
  \medskip

  A \emph{multi-$\mathcal T$-homomorphism} from $A_1, A_2, \dots, A_k$ to $B$ is a morphism $h : \prod_{i=1}^k \underline{A_i} \to \underline{B}$ such that for every term $t \in \mathcal T_k$ and $1 \le i_0 \le n$, the following diagram commutes:
  \[
  \begin{tikzcd}
  \mathcal A_{i_0}^k \times \prodd{\stackrel{1 \le i \le n}{i \neq i_0}}{} \mathcal A_i \arrow[r, "\str"] \arrow[dd, "\mathcal A_{i_0} \brak{t} \times \id"'] & \lp\mathcal A_{i_0} \times \prodd{\stackrel{1 \le i \le n}{i \neq i_0}}{} \mathcal A_i\rp^k \arrow[r, "\cong"] & \lp \prodd{i=1}{n} \mathcal A_i \rp^k \arrow[d, "f^k"] \\
                                                                                                                                                            &                                                                                                              & \mathcal B^k \arrow[d, "\mathcal B \brak{t}"]         \\
  \mathcal A_{i_0} \times \prodd{\stackrel{1 \le i \le n}{i \neq i_0}}{} \mathcal A_i \arrow[r, "\cong"']                                                     & \prodd{i=1}{n} \mathcal A_i \arrow[r, "f"']                                                                   & \mathcal B                                           
  \end{tikzcd}
  \]
\end{definition}

$\triangleright$ We can then define a multi-category structure $\Multi(\mathcal T, \mathcal C)$ over the finite-product multi-category:
\begin{itemize}
  \item the objects are $\mathcal T$-algebra
  \item the multi-morphisms are the multi-$\mathcal T$-homomorphisms
  \item the identities are the identity morphisms
  \item composition is inherited from $\la \mathcal C, \prod, \ra$
\end{itemize}

$\triangleright$ We also define a multi-functor $\underline{-} : \Multi(\mathcal T, \mathcal C) \to \la \mathcal C, \prod \ra$ which sends an algebra to its carrier and a multi-$\mathcal T$-homomorphism to its underlying morphism.

\medskip

$\triangleright$ We define another 2-category $\Cat_\times$ consisting of:
\begin{itemize}
  \item 0-cells are $\la \mathcal C, \prod \ra$, categories with chosen finite products
  \item 1-cells are finite product preserving functors, not necessarily preserving the chosen products
  \item 2-cells are the natural transformations between these functors
\end{itemize}

Therefore, this is a sub-category of $\Cat$.

\medskip

$\triangleright$ We define the data of a 2-functor $\Multi(\mathcal T, -) : \Cat_\times \to \Multicat$ sending:

\subsection{Free extensions}

Free extensions where introduced in \cite{10.1145/3236795}. They are a unifying theory for specifying (generalized) partially-static data-structures by recourse to universal algebra.

\medskip

In this section, let $\pres$ be a presentation, $X$ be a set of variables and $A$ be a $\pres$-algebra.

\begin{definition}[Extension]
  An \emph{extension} of $A$ by $X$ is a triple $\la B, h, e \ra$ where:
  \begin{itemize}
    \item $B$ is a $\pres$-algebra
    \item $h : A \to B$ is a $\pres$-homomorphism
    \item $e : X \to B$ is a function
  \end{itemize}
\end{definition}

\begin{definition}[Morphism of extensions]
  Let $b_1 = \la B_1, h_1, e_1 \ra, b_2 = \la B_2, h_2, e_2 \ra$ be two extensions of $A$ by $X$.

  \medskip

  A \emph{morphism of extensions} $h : b_1 \to b_2$ is a $\pres$-homomorphism $h : B_1 \to B_2$ that makes the following diagram commute:
  \[\begin{tikzcd}
    & {B_1} & \\
    X && A \\
    & {B_2}
    \arrow["h", from=1-2, to=3-2]
    \arrow["{e_1}", from=2-1, to=1-2]
    \arrow["{e_2}"', from=2-1, to=3-2]
    \arrow["{h_1}"', from=2-3, to=1-2]
    \arrow["{h_2}", from=2-3, to=3-2]
  \end{tikzcd}\]
\end{definition}

\begin{definition}[Free extension]
  A \emph{free extension} of $A$ by $X$ is an extension for which there exists a unique morphism of extensions to every other extension of $A$ by $X$.
\end{definition}

\subsection{Distributive combination of theories}

\begin{definition}[Coproduct of signatures]
  Given two signatures $\Sigma_1$ and $\Sigma_2$, we define their coproduct $\Sigma_1 + \Sigma_2$ as the signature with operators the disjoint union of the operators, but retaining their arity:
  $$\Sigma_1 + \Sigma_2 = \{ \iota_i f : n \mid (f : n) \in \Sigma_i, i = 1, 2 \}$$
  Given a $\Sigma_i$-term-in-context $ n \vdash_{\Sigma_i} t$, we denote by $n \vdash_{\Sigma_1 + \Sigma_2} \iota_i [t]$ the $\Sigma_1 + \Sigma_2$-term-in-context obtained by applying the injection $\iota_i$ to all the operators of $t$.

  \medskip

  Similarly, given a $\Sigma_i$-equation in context $eq := (n \vdash_{\Sigma_i} t = t')$, we denote by $\iota_i [eq]$ the $\Sigma_1 + \Sigma_2$-equation in context $n \vdash_{\Sigma_1 + \Sigma_2} \iota_i [t] = \iota_i [t']$.
\end{definition}

\begin{definition}[Distributive combination of theories]
  Let $\mathfrak T : \mathcal O \to \mathcal M$ be a translation from an operad $\mathcal O$ to a theory $\mathcal M$ and $\mathcal A$ be any theory.

  \medskip

  We define the \emph{distributive combination of $\mathfrak T$ over $\mathcal A$} to be the theory $\mathfrak T \triangleleft \mathcal A$ given by

  \begin{align*}
    \Sigma_{\mathfrak T \triangleleft A} &:= \Sigma_M + \Sigma_A\\
    \Ax_{\mathfrak T \triangleleft A} &:= \iota_1[\Ax_\mathcal{M}] \cup \iota_2[\Ax_{\mathcal A}]\\
    & \cup
      \left\{
      \left.
      \begin{array}{c}
      \la x_i\ra_{\stackrel{i=1}{i\neq i_0}}^{n},
      \;
      \mathbf{y} \mid \mathfrak T f(x_1,\ldots,x_{i_0-1},s \mathbf{y},x_{i_0+1},\ldots,x_n)
      \\[4pt]
      \qquad\qquad
      =
      s\left\langle
      \mathfrak T f(x_1,\ldots,x_{i_0-1},\mathbf{y}_j,x_{i_0+1},\ldots,x_n)
      \right\rangle_{j=1}^{|\mathbf{y}|}
      \end{array}
      \;\right|\;
      \begin{array}{l}
      (f:n)\in O_n \\
      (s:|\mathbf{y}|)\in \Sigma_\mathcal{A}
      \end{array}
      \right\}
  \end{align*}
  \label{def:comb_theory}
\end{definition}


\section{Constructing free algebras}

\subsection{Semigroups}
\label{sec:free_semigroups}

\minisection{Free commutative semigroups}

Let's try to construct the free algebra for the presentation of commutative semigroups. We cannot rely on the order of the terms, nor the order of applying the operations.

\medskip

Let $X$ be a set. We consider the theory of commutative semigroups $\CSG$, with a signature $\Sigma$ with a single operator $(+)$ of arity 2, and with the two axioms \begin{align*}
  a, b &\vdash a + b = b + a\\
  a, b, c &\vdash a + (b + c) = (a + b) + c
\end{align*}

We pose $$SX = \{f : X \to \N \mid f \text{ has finite non-empty support}\}$$

For $f, g \in SX$, we pose $$f + g = \lambda x. f\ x + g\ x \in S X$$

We define $\eta_{\CSG} : X \to S X$ for all $x \in X$ by $$\eta_{\CSG} (x) = \lambda y. \begin{cases}
  1 \text{ if } x = y\\
  0 \text{ otherwise}
\end{cases}$$

We pose $F_{\CSG}X = (S X, (+))$ where $+$ is the operator defined above. 

\begin{proposition}
  $F_{\CSG}X$ is a free $\CSG$-model over $X$.
\end{proposition}

\begin{proof}
  \begin{itemize}[label=$\triangleright$]
  \item Let's first check that $F_{\CSG}X$ is indeed a $\CSG$-model. We need to check that the axioms of $\CSG$ are valid in $F_{\CSG}X$. This is straightforward and comes down to the commutativity and associativity of addition in $\N$.
  \item Let $A$ be a $\CSG$-model over $X$ with environment $\phi_A : X \to U\,A$. We want to show that there exists a unique morphism $h$ of $\CSG$-models from $F_{\CSG}X$ to $A$ such that the following diagram commutes:
\[
\begin{tikzcd}
  X \arrow[r, "\eta_{\CSG}"] \arrow[dr, "\phi_A"'] & S X \arrow[d, "h"] \\
  & U\,A
\end{tikzcd}
\]

with $U$ being the forgetful functor from $\CSG$-models to $\Set$.

We define $\Sigma\phi_A : S X \to \underline{A}$ for all $f \in S X$ by $$\Sigma\phi_A(f) = \sum_{x \in X} f\ x \cdot \phi_A(x)$$

We want to show that $\Sigma\phi_A$ is the unique morphism $h$ of $\CSG$-models from $F_{\CSG}X$ to $A$ above. We need to check that $\Sigma\phi_A$ preserves the operator $+$, and that the diagram above commutes.

\begin{itemize}
  \item Preservation of the operator $+$: let $f, g \in S X$. We have:
\begin{align*}
  \Sigma\phi_A(f + g) &= \Sigma\phi_A(\lambda x. f\ x + g\ x)\\
  &= \sum_{x \in X} (f\ x + g\ x) \cdot \phi_A(x)\\
  &= \sum_{x \in X} f\ x \cdot \phi_A(x) + \sum_{x \in X} g\ x \cdot \phi_A(x)\\
  &= \Sigma\phi_A(f) + \Sigma\phi_A(g)
\end{align*}
  \item Commutation of the diagram: let $x \in X$. We have:
\begin{align*}
  (\Sigma\phi_A)(\eta_{\CSG}(x)) &= \Sigma\phi_A \lp \lambda y. \begin{cases}
    1 \text{ if } x = y\\
    0 \text{ otherwise }
  \end{cases} \rp\\
  &= \sum_{y \in X} \begin{cases}
    1 \cdot \phi_A(x) &\text{ if } x = y\\
    0 &\text{ otherwise}
  \end{cases}\\
  &= \phi_A(x)
\end{align*}

Therefore, the diagram commutes.
\item Uniqueness: let $h : F_{\CSG}X \to A$ be a morphism of $\CSG$-models such that the diagram above commutes. We want to show that $h = \Sigma\phi_A$. Let $f \in S X$. We have:
\begin{align*}
  h(f) &= h\lp \sum_{x \in X} f\ x \cdot \eta_{\CSG}(x) \rp\\
  &= \sum_{x \in X} f\ x \cdot h(\eta_{\CSG}(x))  \\
  &= \sum_{x \in X} f\ x \cdot \phi_A(x) & \text{since the diagram commutes} \\
  &= \Sigma\phi_A(f)
\end{align*}
\end{itemize}
\end{itemize}

\medskip

\fbox{Therefore, we proved that $F_{\CSG}X$ is a free $\CSG$-model over $X$.}

\end{proof}

Finally we'll define a multiplication $\mu_\CSG$ so that we have a monad structure $(S, \eta_\CSG, \mu_\CSG)$ on $\Set$. For all $f \in S S X$, we pose $$\mu_\CSG(f) = \lambda x. \sum_{g \in S X} f\ g \cdot g\ x$$

\begin{proposition}
  $(S, \eta_\CSG, \mu_\CSG)$ is a monad on $\Set$.
\end{proposition}

\begin{proof}
  We need to check the monad axioms:
  \begin{itemize}
    \item Left unit: let $f \in S X$. We have:
\begin{align*}
  \mu_\CSG(\eta_\CSG S(f)) &= \mu_\CSG\lp \lambda g. \begin{cases}
    1 \text{ if } g = f\\
    0 \text{ otherwise}
  \end{cases} \rp\\
  &= \lambda x. \sum_{g \in S X} \begin{cases}
    1 \cdot g\ x &\text{ if } g = f\\
    0 &\text{ otherwise}
  \end{cases}\\
  &= \lambda x. f\ x\\
  &= f
\end{align*}
    \item Right unit: let $f \in S X$. We have:
  \begin{align*}
    \mu_\CSG(S\eta_\CSG(f)) &= \mu_\CSG\lp \lambda g. \begin{cases}
      f\ x &\text{ if $g = \eta_\CSG(x)$ for some $x \in X$}\\
      0 &\text{ otherwise}
    \end{cases} \rp\\
    &= \lambda x. \sum_{g \in S X} \begin{cases}
      f\ x \cdot g\ x &\text{ if $g = \eta_\CSG(x)$ for some $x \in X$}\\
      0 &\text{ otherwise}
    \end{cases}\\
    &= \lambda x. f(x) \cdot \eta_\CSG(x)(x)\\
    &= \lambda x. f\ x\\
    &= f
  \end{align*}
    \item Associativity: let $f \in S S S X$. We have:
  \begin{align*}
    \mu_\CSG(S\mu_\CSG(f)) &= \mu_\CSG\lp \lambda g. \sum_{h \in S S X} f\ h \cdot h\ g \rp\\
    &= \lambda x. \sum_{g \in S X}
      \lp\sum_{h \in S S X} f\ h \cdot h\ g \rp \cdot g\ x\\
    &= \lambda x. \sum_{h \in S S X} f\ h \cdot \lp \sum_{g \in S X} h\ g \cdot g\ x \rp\\
    &= \lambda x. \sum_{h \in S S X} f\ h \cdot \mu_\CSG(h)\\
    &= \mu_\CSG(\mu_\CSG S(f))
  \end{align*}
\end{itemize}
\end{proof}

\bigskip

\minisection{Free semigroups}

Compared to the free commutative semigroup, we can rely on the order of the terms. We'll use lists to represent the terms of the free semigroup.

\medskip

Let $X$ be a set. We consider the theory of semigroups $\SG$, with a signature $\Sigma$ with a single operator $(\cdot)$ of arity 2, and with the axiom \begin{align*}
  a, b, c &\vdash a \cdot (b \cdot c) = (a \cdot b) \cdot c
\end{align*}

We pose $$TX = \{l \in X^* \mid l \text{ is finite and non-empty}\}$$

For $f, g \in TX$, we pose $$f \cdot g = f \concat g$$ where $\concat$ is the concatenation of lists.

\medskip

We define $\eta_{\SG} : X \to T X$ for all $x \in X$ by $$\eta_{\SG} (x) = [x]$$

\medskip

We pose $F_{\SG}X = (T X, (\cdot))$ where $\cdot$ is the operator defined above.

\begin{proposition}
  $F_{\SG}X$ is a free $\SG$-model over $X$.
\end{proposition}

\begin{proof}

\begin{itemize}[label=$\triangleright$]
  \item $F_\SG$ is a $\SG$-model: the associativity of $\cdot$ in $F_\SG$ comes from the associativity of list concatenation.
  \item Let $A$ be a $\SG$-model over $X$ with environment $\phi_A : X \to U\,A$.
  
  We define $\Sigma\phi_A : T X \to \underline{A}$ for all $l \in T X$ by $$\Sigma\phi_A(l) = \prod_{x \in l} \phi_A(x)$$ where the product\footnote{product with the operator $\cdot$} is taken in the order of the list $l$, which is well defined because the list is finite, non-empty, and the product is associative. 
  
  Let's show that $\Sigma\phi_A$ is the unique morphism $h$ of $\SG$-models from $F_{\SG}X$ to $A$ such that the following diagram commutes:
\[
\begin{tikzcd}
  X \arrow[r, "\eta_{\SG}"] \arrow[dr, "\phi_A"'] & T X \arrow[d, "h"] \\
  & U\,A
\end{tikzcd}
\]
\begin{itemize}
  \item Preservation of the operator $\cdot$: let $f, g \in T X$. We have:
\begin{align*}
  \Sigma\phi_A(f \cdot g) &= \Sigma\phi_A(f \concat g)\\
  &= \prod_{x \in f \concat g} \phi_A(x)\\
  &= \prod_{x \in f} \phi_A(x) \cdot \prod_{x \in g} \phi_A(x) &\text{by associativity of the product}\\
  &= \Sigma\phi_A(f) \cdot \Sigma\phi_A(g)
\end{align*}
\item Commutation of the diagram: let $x \in X$. We have:
\begin{align*}
  (\Sigma\phi_A)(\eta_{\SG}(x)) &= \Sigma\phi_A([x])\\
  &= \phi_A(x)
\end{align*}
\item Uniqueness: let $h : F_{\SG}X \to A$ be a morphism of $\SG$-models such that the diagram above commutes. We want to show that $h = \Sigma\phi_A$. Let $l \in T X$. We have:
\begin{align*}
  h(l) &= h\lp \prod_{x \in l} \eta_{\SG}(x) \rp\\
  &= \prod_{x \in l} h(\eta_{\SG}(x)) & \text{since $h$ is a morphism}  \\
  &= \prod_{x \in l} \phi_A(x) & \text{since the diagram commutes} \\
  &= \Sigma\phi_A(l)
\end{align*}
\end{itemize}
\end{itemize}

\fbox{Therefore we proved that $F_\SG X$ is a free $\SG$-model over $X$.}

\end{proof}

We'll also define a multiplication $\mu_\SG$ so that we have a monad structure $(T, \eta_\SG, \mu_\SG)$ on $\Set$. For all $l \in T T X$, we pose $$\mu_\SG(l) = \concat_{x \in l} x$$ where the concatenation is taken in the order of the list $l$, which is well defined because the list $l$ is finite, non-empty, and the concatenation is associative.

\begin{proposition}
  $(T, \eta_\SG, \mu_\SG)$ is a monad on $\Set$.
\end{proposition}

\begin{proof}
  We need to check the monad axioms:
  \begin{itemize}
    \item Left unit: let $l \in T X$. We have:
\begin{align*}  
  \mu_\SG(\eta_\SG T(l)) &= \mu_\SG ([l])\\
  &= \concat_{x \in [l]} x\\
  &= l
\end{align*}
    \item Right unit: let $l \in T X$. We have:
  \begin{align*}   
    \mu_\SG(T\eta_\SG(l)) &= \mu_\SG\lp \crochets{[x_i] \mid x_i \in l} \rp\\
    &= \concat_{x \in \crochets{[x_i] \mid x_i \in l}} x\\
    &= \concat_{x_i \in l} [x_i]\\
    &= l
  \end{align*}
    \item Associativity: let $l \in T T T X$. We have:
  \begin{align*}   
    \mu_\SG(T\mu_\SG(l)) &= \mu_\SG\lp \crochets{\concat_{x \in l_i} x \mid l_i \in l} \rp\\
    &= \concat_{x \in \crochets{\concat_{x \in l_i} x \mid l_i \in l}} x\\
    &= \concat_{l_i \in l} \concat_{x \in l_i} x\\
    &= \concat_{x \in \concat_{l_j \in l} l_j} x\\
    &= \concat_{x \in l} x\\
    &= \mu_\SG(\mu_\SG T(l))
  \end{align*}
\end{itemize}
\end{proof}

\subsection{Combining semigroups and commutative semigroups}
\label{sec:comb1}

We will call $\SG \triangleleft \CSG$ the distributive combination of $\SG$ and $\CSG$.

\medskip

The signature of $\SG \triangleleft \CSG$ has two operators: $(\cdot)$ of arity 2, and $(+)$ of arity 2, and the axioms are the union of the axioms of $\SG$ and $\CSG$ : 
\begin{align*}
  a, b, c &\vdash a \cdot (b \cdot c) = (a \cdot b) \cdot c\\
  a, b &\vdash a + b = b + a\\
  a, b, c &\vdash a + (b + c) = (a + b) + c
\end{align*}

We also have the distributivity axioms: \begin{align*}
  a, b, c &\vdash a \cdot (b + c) = a \cdot b + a \cdot c & \text{left distributivity}\\
  a, b, c &\vdash (a + b) \cdot c = a \cdot c + b \cdot c & \text{right distributivity}
\end{align*}

We'll try to combine the two previous examples to hopefully get the free algebra for the presentation.

\medskip

The carrier set of our candidate for the free $\SG \triangleleft \CSG$-model over $X$ is $STX$.

\medskip

We will keep the same definition for the operator $+$ as in the free commutative semigroup, but we'll need to change the definition of the operator $\cdot$ to take into account the fact that we are now working with $STX$ instead of $TX$.

If we write $(p : n) \in f$ when $f(p) = n$, then what we want is $$f \cdot g = \sum_{(p : n) \in f}\sum_{(q : m) \in g} (p \cdot_\SG q : n \cdot m)$$

Formaly, this can be defined for all $f, g \in STX$ as $$f \cdot g = \lambda l : TX. \sum_{l = l_1 \cdot_{\SG} l_2} f(l_1) \cdot g(l_2)$$

\medskip

We define $\eta_\SGCSG : X \to STX = \eta_\CSG \circ \eta_\SG$. Let's first prove that the following diagram commutes:
\[
\begin{tikzcd}
  X \arrow[r, "\eta_\SG"] \arrow[dr, "\eta_\CSG"'] & T X \arrow[r, "\eta_\CSG T"] & S T X\\
  & S X \arrow[ur, "S \eta_\SG"'] 
\end{tikzcd}
\]

Let $x \in X$. We have:
\begin{align*}
  (S \eta_\SG)(\eta_\CSG(x)) &= S \eta_\SG \lp \lambda y. \begin{cases}
    1 \text{ if } x = y\\
    0 \text{ otherwise }
  \end{cases} \rp\\
  &= \lambda y. \begin{cases}
    1 &\text{ if } \eta_\SG(x) = y\\
    0 &\text{ otherwise }
  \end{cases}\\
  &= \eta_\CSG(\eta_\SG(x))
\end{align*}

We pose $F_\SGCSG X = (STX, (\cdot), (+))$, where $(\cdot)$ and $(+)$ are defined above.

\medskip

\begin{proposition}
  \label{prop:F_SGCSG_free}
  $F_\SGCSG X$ is a free $\SG \triangleleft \CSG$-model over $X$.
\end{proposition}

\begin{proof}
  \begin{itemize}[label=$\triangleright$]
  \item Let's first check that $F_\SGCSG X$ is indeed a $\SG \triangleleft \CSG$-model. 
  
  The associativity and commutativity of $(+)$ was already proven for $F_\CSG X$.
  
  For the associativity of $(\cdot)$: let $f, g, h \in STX$. We have:
\begin{align*}
  (f \cdot g) \cdot h &= \lambda l : TX. \sum_{l = l_1 \cdot_{\SG} l_2} (f \cdot g)(l_1) \cdot h(l_2)\\
  &= \lambda l : TX. \sum_{l = l_1 \cdot_{\SG} l_2} \sum_{l_1 = l_{11} \cdot_{\SG} l_{12}} f(l_{11}) \cdot g(l_{12}) \cdot h(l_2)\\
  &= \lambda l : TX. \sum_{l = l_{11} \cdot_{\SG} (l_{12} \cdot_{\SG} l_2)} f(l_{11}) \cdot g(l_{12}) \cdot h(l_2) &\text{by associativity of $\cdot_{\SG}$}\\
  &= \lambda l : TX. \sum_{l = l_{11} \cdot_{\SG} l_3} \sum_{l_3 =l_{12} \cdot_{\SG} l_2} f(l_{11}) \cdot g(l_{12}) \cdot h(l_2)\\
  &= \lambda l : TX. \sum_{l = l_{11} \cdot_{\SG} l_3} f(l_{11}) \cdot (g \cdot h)(l_3)\\
  &= f \cdot (g \cdot h)
\end{align*}

The distributivity of $(\cdot)$ over $(+)$ comes from the distributivity of the product in $\N$ over addition, and from the definition of $(+)$.

Therefore, $F_\SGCSG X$ is a $\SG \triangleleft \CSG$-model.
  \item Let $A$ be a $\SG \triangleleft \CSG$-model over $X$ with environment $\phi_A : X \to U\,A$. We want to show that there exists a unique morphism $h$ of $\SG \triangleleft \CSG$-models from $F_\SGCSG X$ to $A$ such that the following diagram commutes:
\[
\begin{tikzcd}
  X \arrow[r, "\eta_\SGCSG"] \arrow[dr, "\phi_A"'] & S T X \arrow[d, "h"] \\
  & U A
\end{tikzcd}
\]

We define $\Sigma\phi_A : S T X \to \underline{A}$ for all $f \in S T X$ by $$\Sigma\phi_A(f) = \sum_{l \in T X} f\ l \cdot \prod_{x \in l} \phi_A(x)$$

Let's show that $\Sigma\phi_A$ is the unique morphism $h$ of $\SG \triangleleft \CSG$-models from $F_\SGCSG X$ to $A$ above. We need to check that $\Sigma\phi_A$ preserves the operators $(\cdot)$ and $(+)$, and that the diagram above commutes.

\begin{itemize}
  \item Preservation of the operator $(\cdot)$: let $f, g \in S T X$. We have:
\begin{align*}
  \Sigma\phi_A(f \cdot g) &= \Sigma\phi_A(\lambda l : TX. \sum_{l = l_1 \cdot_{\SG} l_2} f(l_1) \cdot g(l_2))\\
  &= \sum_{l \in T X} \lp \sum_{l = l_1 \cdot_{\SG} l_2} f(l_1) \cdot g(l_2) \rp \cdot \prod_{x \in l} \phi_A(x)\\
  &= \sum_{l_1, l_2 \in T X} f(l_1) \cdot g(l_2) \cdot \prod_{x \in l_1 \cdot_{\SG} l_2} \phi_A(x)\\
  &= \sum_{l_1, l_2 \in T X} f(l_1) \cdot g(l_2) \cdot \prod_{x \in l_1} \phi_A(x) \cdot \prod_{x \in l_2} \phi_A(x) &\text{by associativity of the product}\\
  &= \sum_{l_1 \in T X} f(l_1) \cdot \prod_{x \in l_1} \phi_A(x) \cdot \sum_{l_2 \in T X} g(l_2) \cdot \prod_{x \in l_2} \phi_A(x)\\
  &= \Sigma\phi_A(f) \cdot \Sigma\phi_A(g)
\end{align*}
  \item Preservation of the operator $(+)$: let $f, g \in S T X$. We have:
\begin{align*}
  \Sigma\phi_A(f + g) &= \Sigma\phi_A(\lambda l : TX. f\ l + g\ l)\\
  &= \sum_{l \in T X} (f\ l + g\ l) \cdot \prod_{x \in l} \phi_A(x)\\
  &= \sum_{l \in T X} f\ l \cdot \prod_{x \in l} \phi_A(x) + \sum_{l \in T X} g\ l \cdot \prod_{x \in l} \phi_A(x)\\
  &= \Sigma\phi_A(f) + \Sigma\phi_A(g)
\end{align*}
  \item Commutation of the diagram: let $x \in X$. We have:
\begin{align*}
  (\Sigma\phi_A)(\eta_\SGCSG(x)) &= \Sigma\phi_A(\eta_\CSG(\eta_\SG(x)))\\
  &= \Sigma\phi_A \lp \lambda y. \begin{cases}
    1 &\text{ if }\eta_\SG(x) = y\\
    0 &\text{ otherwise }
  \end{cases} \rp\\
  &= \sum_{y \in T X} \begin{cases}
    1 \cdot \prod_{x' \in y} \phi_A(x') &\text{ if } \eta_\SG(x) = y\\
    0 &\text{ otherwise }
  \end{cases}\\
  &= \prod_{x' \in \eta_\SG(x)} \phi_A(x')\\
  &= \phi_A(x)
\end{align*}
  \item Uniqueness: let $h : F_\SGCSG X \to A$ be a morphism of $\SG \triangleleft \CSG$-models such that the diagram above commutes. We want to show that $h = \Sigma\phi_A$. Let $f \in S T X$. We have:
\begin{align*}
  h(f) &= h\lp \sum_{l \in T X} f\ l \cdot \prod_{x \in l} \eta_\SG(x) \rp\\
  &= \sum_{l \in T X} f\ l \cdot \prod_{x \in l} h(\eta_\SG(x)) &\text{since $h$ is a morphism}  \\
  &= \sum_{l \in T X} f\ l \cdot \prod_{x \in l} \phi_A(x) &\text{since the diagram commutes} \\
  &= \Sigma\phi_A(f)
\end{align*}
\end{itemize}

\fbox{Therefore $F_\SGCSG X$ is a free $\SGCSG$-model over $X$.}

\end{itemize}
\end{proof}

We also define a distributive law $\lambda : TSX \to STX$ for all $l \in TSX$ by
$$\lambda ([f_1, \dots, f_n]) = \sum_{(x_1 : n_1) \in f_1} \dots \sum_{(x_n : n_n) \in f_n} \lp [x_1, \cdots, x_n] : \prod_{i=1}^n n_i \rp $$

Formaly, this can be defined for all $l \in TSX$ as

$$\lambda (l) = \lambda x : TX. \begin{cases}
  0 &\text{ if $\abs{x} \neq \abs{l}$}\\
  \prodd{(y, f) \in x \ast l}{} f(y) &\text{ otherwise}
\end{cases}$$ where $x \ast l$ is the elementwise pairing of the list $x$ and the list $l$, which is well defined because $x$ and $l$ have the same length.

\begin{proposition}
  $\lambda$ is a distributive law of the monad $T$ over the monad $S$.
\end{proposition}

\begin{proof}
  We need to check the following axioms:
  \begin{itemize}
    \item $\lambda \circ T \eta_S = \eta_S T$: let $l \in T X$. We have:
\begin{align*}
  (\lambda \circ T \eta_S)(l) &= \lambda(T \eta_S(l))\\
  &= \lambda \lp \crochets{\eta_S(x) \mid x \in l} \rp\\
  &= \lambda x : TX. \begin{cases}
    0 &\text{ if $\abs{x} \neq \abs{l}$}\\
    \prodd{(y, f) \in x \ast \crochets{\eta_S(x) \mid x \in l}}{} f(y) &\text{ otherwise}
  \end{cases}\\
  &= \lambda x : TX. \begin{cases}
    0 &\text{ if $\abs{x} \neq \abs{l}$}\\
    \prodd{(y, z) \in x \ast l}{} \eta_S(z)(y) &\text{ otherwise}
  \end{cases}\\
  &= \eta_S T(l)
\end{align*}
    \item $\lambda \circ \eta_T S = S \eta_T$: let $f \in S X$. We have:
\begin{align*}
  (\lambda \circ \eta_T S)(f) &= \lambda(\eta_T S(f))\\
  &= \lambda(\crochets{f})\\
  &= \lambda x : TX. \begin{cases}
    0 &\text{ if $\abs{x} \neq 1$}\\
    \prodd{(y, g) \in x \ast \crochets{f}}{} g(y) &\text{ otherwise}
  \end{cases}\\
  &= S \eta_T(f)
\end{align*}

\item $\lambda \circ T \mu_S = \mu_S T \circ S \lambda \circ \lambda S$: let $l = [f_1, \dots, f_n] \in T S S X$. We have:
\begin{align*}
  (\lambda \circ T \mu_S)(l) &= \lambda(T \mu_S(l))\\
  &= \lambda\lp \crochets{\mu_S(f_i) \mid i = 1, \dots, n} \rp\\
  &= \lambda x : TX. \begin{cases}
    \prodd{i = 1}{n} \mu_S(f_i)(x_i) &x = [x_1, \dots, x_n]\\
    0 &\text{ otherwise}
  \end{cases}\\
  &= \lambda x : TX. \begin{cases}
    \prodd{i = 1}{n} \sum_{g \in S X} f_i(g) \cdot g(x_i) &x = [x_1, \dots, x_n]\\
    0 &\text{ otherwise}
  \end{cases}\\
  &= \lambda x : TX. \begin{cases}
    \summ{g_1, \dots, g_n \in S X}{} \prodd{i = 1}{n} f_i(g_i) \cdot g_i(x_i) &x = [x_1, \dots, x_m]\\
    0 &\text{ otherwise}
  \end{cases}
\end{align*}

\begin{align*}
  (\mu_S T \circ S \lambda \circ \lambda S)(l) &= \mu_S T(S \lambda(\lambda S(l)))\\
  &= \mu_S T \lp S \lambda \lp \lambda y : TSX. \begin{cases}
    \prodd{i=1}{n} \sum_{g \in S X} f_i(g) \cdot g(y_i) &y = [y_1, \dots, y_n]\\
    0 &\text{ otherwise}
  \end{cases} \rp\rp\\
  &= \mu_S T \lp \lambda h : STX. \summ{g_1, \dots, g_n \in S X \wedge \lambda ([g_1, \dots, g_n]) = h}{} \prodd{i=1}{n} f_i(g_i) \cdot g_i(x_i) \rp\\
  &= \lambda x : TX. \summ{g_1, \dots, g_n \in S X}{} \lp\prodd{i=1}{n} f_i(g_i)\rp \cdot \lambda ([g_1, \dots, g_n])(x)\\
  &= \lambda x : TX. \begin{cases} 
    \summ{g_1, \dots, g_n \in S X}{} \lp\prodd{i=1}{n} f_i(g_i)\rp \cdot \lp \prodd{i=1}{n} g_i(x_i) \rp &x = [x_1, \dots, x_n]\\
    0 &\text{ otherwise}
  \end{cases}
\end{align*}

Therefore, we have $(\lambda \circ T \mu_S)(l) = (\mu_S T \circ S \lambda \circ \lambda S)(l)$.

\item $\lambda \circ \mu_T S = S \mu_T \circ \lambda T \circ T \lambda$: let $l = [[f_{11}, \dots, f_{1m_1}], \dots, [f_{n1}, \dots, f_{nm_n}]] \in T T S X$. We have:
\begin{align*}
  (\lambda \circ \mu_T S)(l) &= \lambda(\mu_T S(l))\\
  &= \lambda(\concat_{i=1}^n [f_{i1}, \dots, f_{im_i}])\\
  &= \lambda x : TX. \begin{cases}
    \prodd{i=1}{n} \prodd{j=1}{m_i} f_{ij}(x_{ij}) &x = \concat_{i=1}^n [x_{i1}, \dots, x_{im_i}]\\
    0 &\text{ otherwise}
  \end{cases}
\end{align*}

\begin{align*}
  &(S \mu_T \circ \lambda T \circ T \lambda)(l)\\
  &= S \mu_T(\lambda T(T \lambda(l)))\\
  &= S \mu_T(\lambda T([\lambda([f_{11}, \dots, f_{1m_1}]), \dots, \lambda([f_{n1}, \dots, f_{nm_n}])]))\\
  &= S \mu_T\lp\lambda T\lp \left[\lambda y : TX. \begin{cases}
    \prodd{j=1}{m_1} f_{1j}(y_j) &y = [y_1, \dots, y_{m_1}]\\
    0 &\text{ otherwise}
  \end{cases}, \dots, \lambda y : TX. \begin{cases}
    \prodd{j=1}{m_n} f_{nj}(y_j) &y = [y_1, \dots, y_{m_n}]\\
    0 &\text{ otherwise}
  \end{cases}\right]\rp\rp\\
  &= S \mu_T\lp \lambda h : STX. \summ{i=1}{n} \summ{y_1, \dots, y_{m_i} \in TX}{} \begin{cases}
    \prodd{j=1}{m_i} f_{ij}(y_j) &h = \lambda y : TX. \begin{cases}
      \prodd{j=1}{m_i} f_{ij}(y_j) &y = [y_1, \dots, y_{m_i}]\\
      0 &\text{ otherwise}
    \end{cases}\\
    0 &\text{ otherwise}
  \end{cases}\rp\\
  &= \lambda x : TX. \begin{cases}
    \summ{i=1}{n} \summ{y_1, \dots, y_{m_i} \in TX}{} \begin{cases}
      \prodd{j=1}{m_i} f_{ij}(y_j) &y_i = [y_{i1}, \dots, y_{im_i}]\\
      0 &\text{ otherwise}
    \end{cases} &x = \concat_{i=1}^n [y_1, \dots, y_{m_i}]\\
    0 &\text{ otherwise}
  \end{cases}\\
  &= \lambda x : TX. \begin{cases}
    \prodd{i=1}{n} \prodd{j=1}{m_i} f_{ij}(x_{ij}) &x = \concat_{i=1}^n [x_{i1}, \dots, x_{im_i}]\\
    0 &\text{ otherwise}
  \end{cases}
\end{align*}

Therefore, we have $(\lambda \circ \mu_T S)(l) = (S \mu_T \circ \lambda T \circ T \lambda)(l)$.
\end{itemize}
\end{proof}

\begin{corollary}
  $(ST, \eta_\SGCSG, \mu_\SGCSG)$ is a monad on $\Set$, where $$\mu_\SGCSG : STST \xrightarrow{S \lambda T} SSTT \xrightarrow{\mu_S\mu_T} ST$$
\end{corollary}


Another approach to get a monad structure on $ST$ is to use monads in extension form. 

\medskip

\cite{Marmolejo2010MonadsAE} shows\footnote{example 8.5} that we can define an $\SG$-algebra $(STX, (-)^\lambda)$ where for all $f : X \to STY$, we have $f^\lambda : TX \to STY$ defined by $f^\lambda([x_1, \dots, x_n]) = \lambda([f(x_1), \dots, f(x_n)])$.  It is then showed that this $\SG$-algebra gives a distributive law of the monad $T$ over the monad $S$, and therefore a monad structure on $ST$. 

\medskip

In extensive form, the monad is $(ST, \eta_\SGCSG, (-)^{(ST)})$ where for all $f : X \to STY$, we have $f^{(ST)} : STX \to STY$ defined by $f^{(ST)} = (f^\lambda)^T$.

We also have that $\lambda = (S \eta_T)^\lambda$ as expected.

\bigskip

We now may want to see whether the monad we got from the distributive law is the same as the one we get from the free $\SG \triangleleft \CSG$-model construction. 

Informally, we want to check whether the following diagram commutes (also see Definition \ref{def:comb_theory}):
\[
\begin{tikzcd}
{\SG, \CSG} \arrow[rr, "\text{combining theory}"] \arrow[dd, "\text{adding monad structure}"'] &                 & \SG \triangleleft \CSG \arrow[dd, "\text{adding monad structure}"] \\
                                                                                               & \stackrel{?}{=} &                                                                    \\
{\Mon(\SG), \Mon(\CSG)} \arrow[rr, "\text{distributive law}"']                                 &                 & \Mon(\SG \triangleleft \CSG)                                      
\end{tikzcd}
\]

The classical monadic structure on $\SGCSG$ comes from the adjunction between $\Set$ and $\SGCSG$-models given by the free $\SGCSG$-model construction. 

\begin{lemma}
  $F_\SGCSG \dashv U$ is an adjunction, where $U : \SGCSG\text{-}\mathbf{Mod} \to \Set$ is the forgetful functor.
\end{lemma}

\begin{proof}
  We need to show that for all $X \in \Set$ and all $\SGCSG$-model $A$, there is a natural bijection between $\SGCSG$-model morphisms from $F_\SGCSG X$ to $A$ and functions from $X$ to $U\,A$. 

  \medskip

  Let $X \in \Set$ and let $A = (\underline{A}, \cdot_A, +_A)$ be a $\SGCSG$-model. We have already shown that there is a unique morphism of $\SGCSG$-models from $F_\SGCSG X$ to $A$ such that the diagram below commutes:
\[
\begin{tikzcd}
  X \arrow[r, "\eta_\SGCSG"] \arrow[dr, "\phi_A"'] & S T X \arrow[d, "\Sigma\phi_A"] \\
  & \underline{A}
\end{tikzcd}
\]
  Therefore, we have a function $\Phi_{X, A}$ from $\SGCSG$-model morphisms from $F_\SGCSG X$ to $A$ to functions from $X$ to $U\,A$ defined by $\Phi_{X, A}(h) = h \circ \eta_\SGCSG$. 

  \medskip

  Conversely, we have a function $\Psi_{X, A}$ from functions from $X$ to $U\,A$ to $\SGCSG$-model morphisms from $F_\SGCSG X$ to $A$ defined by $\Psi_{X, A}(\phi_A) = \Sigma\phi_A$. 

  \medskip

  We need to show that $\Phi_{X, A}$ and $\Psi_{X, A}$ are inverses of each other. 
  
  Let $h : F_\SGCSG X \to A$ be a morphism of $\SGCSG$-models. We have:
  \begin{align*}
    (\Psi_{X, A} \circ \Phi_{X, A})(h) &= \Psi_{X, A}(h \circ \eta_\SGCSG)\\
    &= \Sigma(h \circ \eta_\SGCSG)\\
    &= h
  \end{align*}

  Let $\phi_A : X \to U\,A$ be a function. We have:
  \begin{align*}
    (\Phi_{X, A} \circ \Psi_{X, A})(\phi_A) &= \Phi_{X, A}(\Sigma\phi_A)\\
    &= \Sigma\phi_A \circ \eta_\SGCSG\\
    &= \phi_A
  \end{align*}

  Therefore, $\Phi_{X, A}$ and $\Psi_{X, A}$ are inverses of each other, and we have a natural bijection between $\SGCSG$-model morphisms from $F_\SGCSG X$ to $A$ and functions from $X$ to $U\,A$.
\end{proof}

This adjunction gives us a monad structure on $ST$.

\begin{theorem}
  The monad structure on $ST$ given by the adjunction $F_\SGCSG \dashv U$ is the same as the one given by the distributive law $\lambda$.
\end{theorem}

\begin{proof}
  We'll consider the extensive form of the monads.
  \begin{itemize}[label=$\triangleright$]
    \item \textbf{Carrier sets.}

    Let $X \in \Set$. $UF_\SGCSG X$ is the carrier set of the free $\SGCSG$-model over $X$, which is $STX$. The carrier set of the monad given by the distributive law is also $STX$. Therefore, the carrier sets are the same.
    \item \textbf{Unit.}
    
    The unit is the same for both monads, sending $x \in X$ to the sum of the singleton list $[x]$ in $STX$.
    \item \textbf{Extension.}
    
    Let $f : X \to STY$. Since $F_\SGCSG Y$ is a model of $\SGCSG$, freeness of $F_\SGCSG X$ gives us a unique morphism of $\SGCSG$-models from $F_\SGCSG X$ to $F_\SGCSG Y$ such that the diagram below commutes:
    \[
    \begin{tikzcd}
      X \arrow[r, "\eta_\SGCSG"] \arrow[dr, "f"'] & S T X \arrow[d, "\Sigma f"] \\
      & S T Y
    \end{tikzcd}
    \]
    Therefore, the extension of $f$ in the monad given by the adjunction is $\Sigma f$. The extension of $f$ in the monad given by the distributive law is $f^{(ST)} = (f^\lambda)^T$. We need to show that $\Sigma f = (f^\lambda)^T$.

    \medskip

    Let $f : X \to STY$. Suppose that $$f(x_k) = \summ{j}{}u_{k,j}$$ for some $u_{k, j} \in TY$. Then $$\Sigma f ([x_1, \dots, x_n]) = \summ{j_1, \dots, j_n}{} u_{1,j_1} \cdots u_{n, j_n}$$
    
    On the other hand, we have:
    \begin{align*}
      f^\lambda([x_1, \dots, x_n]) &= \lambda([f(x_1), \dots, f(x_n)])\\
      &= \lambda([ \summ{j_1}{}u_{1,j_1}, \dots, \summ{j_n}{}u_{n, j_n}])\\
      &= \summ{j_1, \dots, j_n}{} u_{1,j_1} \cdots u_{n, j_n}
    \end{align*}
    Therefore, we have $f^\lambda([x_1, \dots, x_n]) = \Sigma f ([x_1, \dots, x_n])$ for all $[x_1, \dots, x_n] \in T X$.

    \medskip

    We can apply $(-)^T$ to both sides of the equation $f^\lambda = \Sigma f$ to get $(f^\lambda)^T = (\Sigma f)^T$. We have $(\Sigma f)^T = \Sigma f$ because $\Sigma f$ is a morphism of $\SGCSG$-models, and therefore preserves the operators of $\SGCSG$. Therefore, we have $f^{(ST)} = (f^\lambda)^T = \Sigma f$.
  \end{itemize}
  This shows that the monad structure on $ST$ given by the adjunction $F_\SGCSG \dashv U$ is the same as the one given by the distributive law $\lambda$.
\end{proof}


\subsection{Monoids}

Now that we have the free semigroup, we can easily construct the free monoid by adding a new constant (arity 0 operator) to the signature, and adding the corresponding axioms.

\medskip

\minisection{Free commutative monoids}

Let $X$ be a set. We consider the theory of commutative monoids $\CM$, with a signature $\Sigma$ with two operators: $(+)$ of arity 2, and $0$ of arity 0, and with the three axioms \begin{align*}
  a, b &\vdash a + b = b + a\\
  a, b, c &\vdash a + (b + c) = (a + b) + c\\
  a &\vdash a + 0 = a = 0 + a
\end{align*}

We pose\footnote{we may want to find a different name, as it may be confused with the one for the commutative semigroup} $$SX = \{f : X \to \N \mid f \text{ has finite support}\}$$ 

There is no non-empty support condition as in the free commutative semigroup, which means that the function $f$ can be the constant 0 function. This corresponds to the new constant $0$ in the signature.

\medskip

We pose $$\mathbf{0} = \lambda x. 0 \in S X$$.

\medskip

We keep the same definition for the operator $+$ and for $\eta_{\CM}$ as in the free commutative semigroup.

\medskip

We pose $F_{\CM}X = (S X, (+), \mathbf{0})$ where $+$ and $\eta_{\CM}$ are defined as above.

We want to show that $F_{\CM}X$ is a free $\CM$-model over $X$.

\begin{itemize}[label=$\triangleright$]
  \item $F_{\CM}X$ is a $\CM$-model: the axioms of $\CM$ are valid in $F_{\CM}X$ because of the commutativity and associativity of addition in $\N$, and because of the existence of the constant function $\mathbf{0}$ which acts as the identity for $+$.
  \item Let $A = (\underline{A}, +_A, 0_A)$ be a $\CM$-model over $X$ with environment $\phi_A : X \to U\,A$. We define $\Sigma\phi_A : S X \to \underline{A}$ for all $f \in S X$ by $$\Sigma\phi_A(f) = \sum_{x \in X} f\ x \cdot \phi_A(x)$$
  
  We have $\Sigma\phi_A(\mathbf{0}) = 0_A$.

  For the sake of doing the proof at least once, we'll prove that $\Sigma\phi_A$ is well defined on $S X \setminus \{\mathbf{0}\}$.

  \begin{proposition}
    For all $f \in S X$, the sum $\sum_{x \in X} f\ x \cdot \phi_A(x)$ is well defined.
  \end{proposition}

  \begin{proof}
    Let $f \in S X \setminus \{\mathbf{0}\}$. We note $\Supp f = \{x \in X \mid f(x) \neq 0\}$.
    
    Since $f$ has finite support, $\Supp f$ is a finite set. Moreover, since $f \neq \mathbf{0}$, $\Supp f$ is non-empty. Therefore, we can write $\Supp f = \{x_1, x_2, ..., x_n\}$ for some $n \geq 1$ and we have a bijection $\sigma$ between $\{1, 2, ..., n\}$ and $\Supp f$ given by $i \mapsto x_i$. 
    
    Therefore, we can rewrite the sum as follows:
    \begin{align*}
      \sum_{x \in X} f\ x \cdot \phi_A(x) &= \sum_{x \in \Supp f} f\ x \cdot \phi_A(x)\\
      &= \sum_{i=1}^n f\ \sigma(i) \cdot \phi_A(\sigma(i))
    \end{align*}
    The definition is valid because $+_A$ is associative.

    Let $\sigma'$ be another bijection between $\{1, 2, ..., n\}$ and $\Supp f$. Let's show that we get the same value for the sum if we use $\sigma'$ instead of $\sigma$. 
    This is true because $\sigma$ and $\sigma'$ are bijections, so there exists a permutation $\pi$ of $\{1, 2, ..., n\}$ such that $\sigma' = \sigma \circ \pi$. Therefore, we have:
    \begin{align*}
      \sum_{i=1}^n f\ \sigma' (i) \cdot \phi_A(\sigma'(i)) &= \sum_{i=1}^n f\ \sigma(\pi(i)) \cdot \phi_A(\sigma(\pi(i)))\\
      &= \sum_{i=1}^n f\ \sigma(i) \cdot \phi_A(\sigma(i)) &\text{by commutativity of $+_A$}
    \end{align*}

    Therefore, the definition of $\Sigma\phi_A(f)$ does not depend on the choice of the bijection between $\{1, 2, ..., n\}$ and $\Supp f$, and is well defined.

    Now if $f = \mathbf{0}$, we have $\Supp f = \varnothing$, and the sum is the empty sum, which is defined to be $0_A$.
  \end{proof}

  We can now prove that $\Sigma\phi_A$ is the unique morphism $h$ of $\CM$-models from $F_{\CM}X$ to $A$ such that the following diagram commutes:
\[
\begin{tikzcd}
  X \arrow[r, "\eta_{\CM}"] \arrow[dr, "\phi_A"'] & S X \arrow[d, "h"] \\
  & U\,A
\end{tikzcd}
\]
\begin{itemize}
  \item Preservation of the operator $+$: let $f, g \in S X$. We have:
\begin{align*}
  \Sigma\phi_A(f + g) &= \Sigma\phi_A(\lambda x. f\ x + g\ x)\\
  &= \sum_{x \in X} (f\ x + g\ x) \cdot \phi_A(x) &\text{by definition}\\
  &= \sum_{x \in X} f\ x \cdot \phi_A(x) + \sum_{x \in X} g\ x \cdot \phi_A(x) & \text{since both $f$ and $g$ have finite support}\\
  &= \Sigma\phi_A(f) + \Sigma\phi_A(g)
\end{align*}
  \item Commutation of the diagram: $\eta_\CM = \eta_\CSG$, so the proof is the same as for the free commutative semigroup.
  \item Uniqueness: let $h : F_{\CM}X \to A$ be a morphism of $\CM$-models such that the diagram above commutes. We want to show that $h = \Sigma\phi_A$. Let $f \in S X$. We'll show this lemma first:
  
  \begin{lemma}
  For all $f \in S X$, we have $f = \sum_{x \in X} f\ x \cdot \eta_{\CM}(x)$.
  \end{lemma}

  \begin{proof}
    Let $f \in S X$. We have $\Supp f = \{x \in X \mid f(x) \neq 0\}$. Since $f$ has finite support, $\Supp f$ is a finite set. Moreover, since $f$ is not necessarily non-zero, $\Supp f$ can be empty. 

    If $\Supp f$ is empty, then $f = \mathbf{0}$ and we have:
    $$\sum_{x \in X} f\ x \cdot \eta_{\CM}(x) = \sum_{x \in X} 0 \cdot \eta_{\CM}(x) = 0 = f$$

    If $\Supp f$ is non-empty, we can write $\Supp f = \{x_1, x_2, ..., x_n\}$ for some $n \geq 1$. We'll prove by induction on $n$ that $f = \sum_{x \in X} f\ x \cdot \eta_{\CM}(x)$.
    \begin{itemize}
      \item If $n = 1$, then $\Supp f = \{x_1\}$ and we have $f = f\ x_1 \cdot \eta_{\CM}(x_1) = \sum_{x \in X} f\ x \cdot \eta_{\CM}(x)$.
      \item If $n > 1$, we have $\Supp f = \{x_1, x_2, ..., x_n\}$. 
      
      We can write $f = f\ x_1 \cdot \eta_{\CM}(x_1) + f'$ where $f' = \lambda x. \begin{cases}
        0 &\text{ if } x = x_1\\
        f\ x &\text{ otherwise}
      \end{cases}$


      We have $\Supp f' = \{x_2, ..., x_n\}$, so by the induction hypothesis, we have $f' = \sum_{x \in X} f'\ x \cdot \eta_{\CM}(x)$. Therefore, we have:
      \begin{align*}
        f &= f\ x_1 \cdot \eta_{\CM}(x_1) + f'\\
        &= f\ x_1 \cdot \eta_{\CM}(x_1) + \sum_{x \in X} f'\ x \cdot \eta_{\CM}(x)\\
        &= f\ x_1 \cdot \eta_{\CM}(x_1) + \sum_{x \in X} f\ x
        \cdot \eta_{\CM}(x) - f\ x_1 \cdot \eta_{\CM}(x_1) &\text{by definition of $f'$}\\
        &= \sum_{x \in X} f\ x \cdot \eta_{\CM}(x)
      \end{align*}
    \end{itemize}
  \end{proof}
  
  We then have:
\begin{align*}
  h(f) &= h\lp \sum_{x \in X} f\ x \cdot \eta_{\CM}(x) \rp\\
  &= \sum_{x \in X} f\ x \cdot h(\eta_{\CM}(x)) \\
  &= \sum_{x \in X} f\ x \cdot \phi_A(x) & \text{since the diagram commutes} \\
  &= \Sigma\phi_A(f)
\end{align*}
\end{itemize}
\end{itemize}

\fbox{Therefore we proved that $F_\CM X$ is a free $\CM$-model over $X$.}


\subsection{Distributive combination of semigroups and commutative semigroups}

Let's first start with a case of study. We'll take $\mathcal A$ to be commutative semigroups and $\mathcal M$ to be semigroups. We denote by $(+)$ the operator on $\mathcal A$, and $(\cdot)$ the operator on $\mathcal M$.

\medskip

We define $\mathcal O$ by $\mathcal O_0 = \varnothing$ and $\mathcal O_n = \{ x_1, \dots, x_n \vdash x_1 \cdots x_n \}$ for $n \ge 1$. We will simply write $\mathcal O_n = \{\mu_n\}$.

\begin{proposition}
  $\mathcal O$ is the smallest operad containing\footnote{we may say that $\mathcal O$ is the operad spanned by $(\cdot)$} $(\cdot)$.
\end{proposition}

\begin{proof}
  \begin{itemize}[label=$\triangleright$]
    \item \textbf{$\mathcal O$ is an operad.} 
    
    associativity comes from the associativity of $(\cdot)$ and the neutrality of the identity is trivial.
    \item \textbf{minimality condition.} 
    
    Let $\mathcal O'$ be an operad containing $(\cdot)$. We want to show that $\mathcal O$ is a sub-operad of $\mathcal O'$. We have $\mu_2 \in \mathcal O'_2$ since $\mathcal O'$ contains $(\cdot)$. We can then construct $\mu_n$ for all $n \ge 1$ by composition of $\mu_2$ with itself, and using the identity for the case $n=1$. Therefore, we have $\mu_n \in \mathcal O'_n$ for all $n \ge 1$, which means that $\mathcal O$ is a sub-operad of $\mathcal O'$.
  \end{itemize}
\end{proof}

We then define a translation $\mathfrak T : \mathcal O \to \mathcal M$ by interpreting each operator as the $n$-ary multiplication, i.e $\mathfrak T \mu_n (x_1, \dots, x_n) = x_1\cdots x_n$.

\begin{proposition}
  $\mathfrak T : \mathcal O \to \mathcal M$ is a surjective translation.
\end{proposition}

\begin{proof}
  \begin{itemize}[label=$\triangleright$]
    \item \textbf{$\mathfrak T$ is a translation.}
    \begin{itemize}
      \item the identity condition is trivialy true.
      \item the translation respects composition thanks to the associativity of $(\cdot)$.
    \end{itemize}
    \item \textbf{$\mathfrak T$ is surjective.}
    
    Let \(t \in \mathcal M_n\). Since \(\mathcal M\) is the algebraic theory of semigroups, \(t\) is represented by a nonempty word in the variables \(x_1,\dots,x_n\). Thus we may write
    \[
    t = x_{i_1}x_{i_2}\cdots x_{i_m}
    \]
    for some \(m \ge 1\) and indices \(i_j \in \{1,\dots,n\}\).

    Consider the unique \(m\)-ary operation \(\mu_m \in \mathcal O_m\). By definition of the translation,
    \[
    \mathfrak T(\mu_m)=x_1x_2\cdots x_m.
    \]

    We define a renaming $\rho : m \to n$
    by
    \[
    \rho(j)=i_j.
    \]

    Applying this renaming to \(\mathfrak T(\mu_m)\) gives
    \[
    \mathfrak T(\mu_m)[\rho]
    =
    (x_1x_2\cdots x_m)[\rho]
    =
    x_{\rho(1)}x_{\rho(2)}\cdots x_{\rho(m)}
    =
    x_{i_1}x_{i_2}\cdots x_{i_m}
    =
    t.
    \]

    Therefore every term \(t \in \mathcal M_n\) is of the form $t=\mathfrak T \mu_m[\rho]$ for some renaming \(\rho : m \to n\). 
    
    Hence the translation $\mathfrak T : \mathcal O \to \mathcal M$ is surjective.
  \end{itemize}
\end{proof}

Let's give an explicit description of the theory $\mathfrak T \triangleleft \mathcal A$.

\begin{itemize}
  \item \textbf{Signature.} The signature $\Sigma_{\mathfrak T \triangleleft \mathcal A}$ contains the operators of $\mathcal M$ and the operators of $\mathcal A$. Therefore, it contains the operator $(\cdot)$ of arity 2 and the operator $(+)$ of arity 2.
  \item \textbf{Axioms.}
  \begin{align*}
    &\text{$\mathcal M$ axioms}\\
    x, y, z &\vdash x \cdot (y \cdot z) = (x \cdot y) \cdot z\\
    &\text{$\mathcal A$ axioms}\\
    x, y &\vdash x + y = y + x\\
    x, y, z &\vdash x + (y + z) = (x + y) + z\\
    &\text{distributivity axioms}\\
    x, y, z &\vdash x \cdot (y + z) = x \cdot y + x \cdot z\\
    x, y, z &\vdash (x + y) \cdot z = x \cdot z + y \cdot z
  \end{align*}
\end{itemize}

We define the multi-functor $\underline{-} : \Multi(\pres, \mathcal U) \to \la \mathcal U, \prod \ra$ such that the following diagram commutes:
\[
\begin{tikzcd}
	{\Multi(\pres, \mathcal U)} & {\la \mathcal U, \prod \ra} \\
	{\Mod(\mathcal T, \mathcal U)} & {\mathcal U}
	\arrow["{\underline{-}}", from=1-1, to=1-2]
	\arrow["{-_o}"', maps to, from=1-1, to=2-1]
	\arrow["{-_o}", maps to, from=1-2, to=2-2]
	\arrow["U"', from=2-1, to=2-2]
\end{tikzcd}
\]
This is the anaology in $\Multicat$ of the forgetful functor in $\Cat$, i.e the forgetful multi-functor.

\begin{proposition}
  \label{prop:forgetful_multi_comm}
  Let $\mathcal T$ be a theory. $\underline{-} : \Multi(\pres, \Set) \to \la \Set, \prod \ra$ has a left adjoint if and only if $\mathcal T$ is commutative.
\end{proposition}

\begin{proof}
  See section 6 of Ohad's notes.
\end{proof}

For vertices:
\begin{itemize}
  \item $\Mod(\mathfrak T \triangleleft \mathcal A, \Set)$ is the category of $\mathfrak T \triangleleft \mathcal A$-models in $\Set$ (i.e with carrier sets in $\Set$), i.e the category of sem-rgs.
  \item $\Mod(\mathcal M, \Set)$ is the category of $\mathcal M$-models in $\Set$, i.e the category of semigroups.
  \item $\Mod(\mathcal O, \la \Set, \prod \ra)$ is the category of $\mathcal O$-models in $\la \Set, \prod \ra$. This category coincides with the category of semigroups in $\Set$ (cf. example 9 of Ohad's notes).
  \item $\Mod(\mathcal O, \Multi(\mathcal A, \Set))$ is the category of $\mathcal O$-models in $\Multi(\mathcal A, \Set)$. 
  
  $\Multi(\mathcal A, \Set)$ is the multicategory where:
  \begin{itemize}
    \item objects are $\mathcal A$-models in $\Set$, i.e commutative semigroups.
    \item multi-morphisms are multi-$\mathcal A$-homomorphisms. It can be showed\footnote{maybe do it?} that multi-$\mathcal A$-homomorphisms are multi-linear.
  \end{itemize}
  Similarly to the previous case, $\Mod(\mathcal O, \Multi(\mathcal A, \Set))$ coincides with the category of sem-rgs in $\Set$. The the distributivity axioms hold in $\Mod(\mathcal O, \Multi(\mathcal A, \Set))$ thanks to the multi-linearity of the multi-morphisms in $\Multi(\mathcal A, \Set)$.
\end{itemize}

For arrows:
\begin{itemize}
  \item $\underline{-}_{\mathcal O \triangleleft \mathcal A}$ is such that for $A = (UA, (+), (\cdot)) \in \Mod(\mathfrak T \triangleleft \mathcal A, \Set)$, $\ul{A}_{\mathcal O \triangleleft \mathcal A}$ consists of:
  \begin{itemize}
    \item $\ul A = (UA, (+))$ the $\mathcal A$-model given by the interpretation of the operators of $\mathcal A$ in $A$.
    \item $\ul{A}_{\mathcal O \triangleleft \mathcal A} \brak{\mu_n} (x_1, \dots, x_n) = x_1 \cdot\ \cdots\ \cdot x_n \in \ul A$
  \end{itemize}

  \item $\underline{-}_{\mathcal M \triangleleft}$ is the functor sending a $\mathfrak T \triangleleft \mathcal A$-model $M$ to the $\mathcal M$-model in $\Set$ given by the interpretation of the operators of $\mathcal M$ in $M$, i.e the semigroup structure of $M$.
  
  For $A = (\underline{A}, (+), (\cdot)) \in \Mod(\mathfrak T \triangleleft \mathcal A, \Set)$, we have $\underline{A}_{\mathcal M \triangleleft} = (\underline{A}, (\cdot))$. It is easy to check that this is indeed a $\mathcal M$-model in $\Set$ because the axioms of $\mathcal M$ are included in the axioms of $\mathfrak T \triangleleft \mathcal A$.
  \item $\Mod(\mathfrak T, \Set)$ is defined in Proposition~\ref{prop:mod_translation_functor}. We have $$\lp \Mod(\mathfrak T, \la \mathcal C, \prod \ra) A \rp \brak{f} := A \brak{\mathfrak T f} \qquad \Mod(\mathfrak T, \la \mathcal C, \prod \ra) h := h$$
  \item $\Mod(\mathcal O, \underline{-})$ is defined in Proposition~\ref{prop:mod_2_functor}. It forgets about the model structure for $\mathcal A$, keeping only its carrier set, therefore we are left with a $\mathcal O$-model in $\Set$.
\end{itemize}

\begin{remarque}
  $\Mod(\mathfrak T, \Set)$ is therefore an isomorphism of categories.
\end{remarque}

\begin{proposition}
  The square above is a pullback square.
\end{proposition}

\begin{proof}
  Informaly, this is true because a model of $\mathfrak T \triangleleft \mathcal A$ consists of a model of $\mathcal M$ and a model of $\mathcal A$ such that the operations of $\mathcal O$ distribute over the operations of $\mathcal A$. Conversely, we can reconstruct a model for the combination with both those models.
  
  \medskip

  We'll give a constructive proof anyway.

  \begin{itemize}[label=$\triangleright$]
    \item \textbf{commutativity.}
    
    Let $A \in \Mod(\mathfrak T \triangleleft \mathcal A, \Set)$. 
    
    We clearly have $\underline{\Mod(\mathfrak T, \Set)(\underline{A}_{\mathcal M \triangleleft})} = \underline{\Mod(\mathcal O, \underline{-})(\underline{A}_{\mathcal O \triangleleft \mathcal A})}$, since both sides are equal to $\underline{A}$.

    Moreover, for $f \in \mathcal O_n$, we have:
    \begin{align*}
      \Mod(\mathfrak T, \Set)(\underline{A}_{\mathcal M \triangleleft}) \brak{f} &= \underline{A}_{\mathcal M \triangleleft} \brak{\mathfrak T f} &\text{by definition of $\Mod(\mathfrak T, \Set)$}\\
      &= \underline{A}_{\mathcal M \triangleleft} \brak{\mu_n} &\text{since $\mathcal O$ is spanned by $\mu_n$}\\
      &= \underline{A}_{\mathcal O \triangleleft \mathcal A} \brak{\mu_n} &\text{by definition of $\underline{-}_{\mathcal O \triangleleft \mathcal A}$}\\
      &= \Mod(\mathcal O, \underline{-})(\underline{A}_{\mathcal O \triangleleft \mathcal A}) \brak{f} &\text{by definition of $\Mod(\mathcal O, \underline{-})$}
    \end{align*}
    \item \textbf{pullback.}
    
    Let $\mathcal C$ be a category, $F : \mathcal C \to \Mod(\mathcal M, \Set)$ and $G : \mathcal C \to \Mod(\mathcal O, \la \Set, \prod \ra)$ be two functors such that $\Mod(\mathfrak T, \Set) \circ F = \Mod(\mathcal O, \underline{-}) \circ G$. We want to show that there exists a unique functor $H : \mathcal C \to \Mod(\mathfrak T \triangleleft \mathcal A, \Set)$ such that the following diagram commutes:
    \[
    % https://q.uiver.app/#q=WzAsNSxbMSwxLCJcXE1vZChcXG1hdGhmcmFrIFQgXFx0cmlhbmdsZWxlZnRcXG1hdGhjYWwgQSwgXFxTZXQpIl0sWzEsMiwiXFxNb2QoXFxtYXRoY2FsIE0sIFxcU2V0KSJdLFsyLDIsIlxcTW9kKFxcbWF0aGNhbCBPLCBcXGxhIFxcU2V0LCBcXHByb2QgXFxyYSkiXSxbMiwxLCJcXE1vZChcXG1hdGhjYWwgTywgXFxNdWx0aShcXG1hdGhjYWwgQSwgXFxTZXQpKSJdLFswLDAsIlxcbWF0aGNhbCBDIl0sWzMsMiwiXFxNb2QoXFxtYXRoY2FsIE8sIFxcdW5kZXJsaW5ley19KSJdLFswLDMsIlxcdW5kZXJsaW5ley19X3tcXG1hdGhjYWwgTyBcXHRyaWFuZ2xlbGVmdCBcXG1hdGhjYWwgQX0iXSxbMSwyLCJcXE1vZChcXG1hdGhmcmFrIFQsIFxcU2V0KSIsMl0sWzAsMSwiXFx1bmRlcmxpbmV7LX1fe1xcbWF0aGNhbCBNfSIsMl0sWzQsMywiRyIsMCx7ImN1cnZlIjotM31dLFs0LDEsIkYiLDIseyJjdXJ2ZSI6M31dLFs0LDAsIkgiLDEseyJzdHlsZSI6eyJib2R5Ijp7Im5hbWUiOiJkYXNoZWQifX19XSxbMCw3LCIiLDEseyJsZXZlbCI6MSwic3R5bGUiOnsibmFtZSI6ImNvcm5lciJ9fV1d&macro_url=https%3A%2F%2Fraw.githubusercontent.com%2FS-c-r-a-t-c-h-y%2Fsemiring-solvers%2Frefs%2Fheads%2Fmain%2Fmacros.sty
    \begin{tikzcd}
      {\mathcal C} && \\
      & {\Mod(\mathfrak T \triangleleft\mathcal A, \Set)} & {\Mod(\mathcal O, \Multi(\mathcal A, \Set))} \\
      & {\Mod(\mathcal M, \Set)} & {\Mod(\mathcal O, \la \Set, \prod \ra)}
      \arrow["H"{description}, dashed, from=1-1, to=2-2]
      \arrow["G", curve={height=-18pt}, from=1-1, to=2-3]
      \arrow["F"', curve={height=18pt}, from=1-1, to=3-2]
      \arrow["{\underline{-}_{\mathcal O \triangleleft \mathcal A}}", from=2-2, to=2-3]
      \arrow["{\underline{-}_{\mathcal M}}"', from=2-2, to=3-2]
      \arrow["{\Mod(\mathcal O, \underline{-})}", from=2-3, to=3-3]
      \arrow[""{name=0, anchor=center, inner sep=0}, "{\Mod(\mathfrak T, \Set)}"', from=3-2, to=3-3]
    \end{tikzcd}
    \]
    We define $H : \mathcal C \to \Mod(\mathfrak T \triangleleft \mathcal A, \Set)$ as follows: for $c \in \mathcal C$, we have $H c = (U F c, (+_c), (\cdot_c))$ where:
    \begin{itemize}
      \item $U F c$ is the underlying set of the $\mathcal M$-model $F c$.
      \item $(+_c)$ is the binary operation on $U F c$ given by the interpretation of the operator $(+)$ of $\mathcal A$ in $G c$, i.e $(+_c) = G c \brak{+}$.
      \item $(\cdot_c)$ is the binary operation on $U F c$ given by the interpretation of the operator $(\cdot)$ of $\mathcal M$ in $F c$, i.e $(\cdot_c) = F c \brak{\cdot}$.
    \end{itemize}
    We then have to check that $H c$ is indeed a $\mathfrak T \triangleleft \mathcal A$-model, i.e that the axioms of $\mathfrak T \triangleleft \mathcal A$ hold in $H c$. This is true because:
    \begin{itemize}
      \item the axioms of $\mathcal M$ hold in $H c$ since $F c$ is a $\mathcal M$-model and the interpretation of the operators of $\mathcal M$ in $H c$ is the same as the interpretation of the operators of $\mathcal M$ in $F c$.
      \item the axioms of $\mathcal A$ hold in $H c$ since $G c$ is a $\mathcal O$-model in $\Multi(\mathcal A, \Set)$ and the interpretation of the operators of $\mathcal A$ in $H c$ is the same as the interpretation of the operators of $\mathcal A$ in $G c$.
      \item the distributivity axioms hold in $H c$ since $G c$ is a $\mathcal O$-model in $\Multi(\mathcal A, \Set)$ and the interpretation of the operators of $\mathcal M$ in $H c$ is the same as the interpretation of the operators of $\mathcal M$ in $F c$, and the interpretation of the operators of $\mathcal A$ in $H c$ is the same as the interpretation of the operators of $\mathcal A$ in $G c$, and the distributivity axioms hold in $\Multi(\mathcal A, \Set)$.
    \end{itemize}

\end{proof}

\subsection{Free algebras for the combination}

\label{sec:free_alg_ex}

We continue with the example of the previous section and we now try to construct the free algebra for $\mathfrak T \triangleleft \mathcal A$ using the free algebras for $\mathcal M$ and $\mathcal A$, similarly to Section~\ref{sec:comb1}, but in a more abstract way. We fix a set $X$.

\medskip

We'll use the results of Section~\ref{sec:free_semigroups}, and we write\footnote{acutally that's not correct since we need to define functors} $F_\mathcal{M} = F_\SG$ and $F_\mathcal{A} = F_\CSG$.

\medskip

\textbcf{Step 1.} 

Let $A_0 := F_\mathcal{M} X \models \mathcal M$ be the free $\mathcal M$-algebra over $X$. $$\underline{A_0} = TX \qquad A_0\brak{(\cdot)} = (\concat) \qquad \eta^{A_0}(x) = [x]$$

\textbcf{Step 2.} 

We apply the functor $$\underline{-}_\mathfrak{T} := \Mod(\mathfrak T, \Set) : \Mod(\mathcal M, \Set) \to \Mod(\mathcal O, \la \Set, \prod \ra)$$ to $A_0$. This gives $A_1 := \underline{A_0}_\mathfrak{T} \models \mathcal O$.

$$\underline{A_1} = \underline{A_0} = TX \qquad A_1 \brak{\mu_n}(x_1, \dots, x_n) = x_1 \concat \cdots \concat x_n \qquad \eta_{A_0}^{A_1} = \id$$

\textbcf{Step 3.} 

By Proposition~\ref{prop:forgetful_multi_comm}, we have an adjunction $$F_\mathcal{A} \dashv \underline{-} : \Multi(\mathcal A, \Set) \to \la \Set, \prod \ra$$ in $\Multicat$ since $\mathcal A$ is commutative. Applying the 2-functor $\Mod(\mathcal O, -) : \Multicat \to \Cat$ gives a left adjoint $$F_{\mathcal O \to \mathcal A} : \Mod(\mathcal O, F_\mathcal{A}) : \Mod(\mathcal O, \la \Set, \prod \ra) \to \Mod (\mathcal O, \Multi(\mathcal A, \Set))$$

We pose $A_2 := F_{\mathcal O \to \mathcal A} A_1 \models \mathcal O \ast \mathcal A$.

$$\underline{A_2} = STX \qquad \eta_{A_1}^{A_2} (x) = \sum_{w \in TX} \delta_x^w \cdot w$$

$F_\mathcal{A}$ extends the morphisms in $\la \Set, \prod \ra$ multi-linearly. Therefore if $a_i = \summ{w \in TX}{} n_w^{(i)} \cdot w \in STX$, then 
\begin{align*}
  A_2\brak{\mu_n} (a_1, \dots, a_n) &= \sum_{w_1, \dots, w_n \in TX} (n_{w_1}^{(1)} \cdots n_{w_n}^{(n)}) \cdot (A_1 \brak{\mu_n} (w_1, \dots, w_n))\\
  &= \sum_{w_1, \dots, w_n \in TX} (n_{w_1}^{(1)} \cdots n_{w_n}^{(n)}) \cdot (w_1 \concat \cdots \concat w_n)
\end{align*}

Lemma 33 of Ohad's notes then is just saying that any $x \in STX$ can be written as $$x = \sum_{i=1}^n w_i$$ for some $(w_i) \in (TX)^n$, which is no surprise.

\medskip

\textbcf{Step 4.}

Let $A_3 := \Mod(\mathcal O, \underline{-}) A_2 \models O$ be the underlying operadic algebra. 

$$\underline{A_3} = \underline{A_2} = STX \qquad A_3 \brak{\mu_n} = A_2 \brak{\mu_n} \qquad \eta_{A_2}^{A_3} = \id$$

For each $t \in \mathcal M_n$, we use surjectivity to get some $f \in \mathcal O_m$ and $\theta : m \to n$ such that $t = \mathfrak T f [\theta]$. Then $A_\mathcal{M} \brak t e := A_3 \brak f e \circ \theta$.

\begin{lemma}
  \label{lem:A_M_ind_mod}
  With the preceding notation:
  \begin{itemize}
    \item the interpretation $A_\mathcal{M} \brak{t} e$ is independent of the decomposition $t = \mathfrak T f [\theta]$
    \item the resulting algebra validates $\mathcal M$, i.e $A_\mathcal{M} \models \mathcal M$
  \end{itemize}
\end{lemma}

\begin{proof}
  We will show that, given $f_i \in \mathcal O_{n_i}, \theta_i : n_i \to m$, if $m \vdash_\mathcal{M} \mathfrak T f_1 [\theta_1] = \mathfrak T f_2 [\theta_2]$, then for all $e : [m] \to \underline{A_\mathcal{M}} = \underline{A_3} = \underline{A_2}$, we have $$A_2 \brak{f_1} (e \circ \theta_1) = A_2 \brak{f_2} (e \circ \theta_2)$$
  For $r \in [m]$, let $e(r) = \summ{w \in TX}{} n_w^r \cdot w$. Then
  \begin{align*}
    A_2 \brak{f_1} (e \circ \theta_1) &= A_2 \brak{f_1}  \la e(\theta_1(j)) \ra_{j=1}^m &\text{evaluation in an operadic algebra}\\
    &= A_2 \brak{f_1}  \la \summ{w \in TX}{} n_w^{\theta_1(j)} \cdot w \ra_{j=1}^m &\text{substitution}\\
    &= \sum_{w_1, \dots, w_{n_1} \in TX} (n_{w_1}^{\theta_1(1)} \cdots n_{w_{n_1}}^{\theta_1(m)}) \cdot (A_1 \brak{f_1} (w_1, \dots, w_{n_1})) & \Mod(\mathcal O, F_\mathcal{A})\text{-def}\\
    &= \sum_{w_1, \dots, w_{n_1} \in TX} (n_{w_1}^{\theta_1(1)} \cdots n_{w_{n_1}}^{\theta_1(m)}) \cdot (A_0 \brak{\mathfrak T f_1}(w_1, \dots, w_{n_1})) &\underline{-}_\mathfrak{T}\text{-def}\\
    &= \sum_{w_1, \dots, w_{n_2} \in TX} (n_{w_1}^{\theta_2(1)} \cdots n_{w_{n_2}}^{\theta_2(m)}) \cdot (A_0 \brak{\mathfrak T f_2}(w_1, \dots, w_{n_2})) &\text{by hypothesis}\\
    &= \dots &\text{reverse steps}\\
    &= A_2 \brak{f_2} (e \circ \theta_2)
  \end{align*}
  This proves the first point. The second point is then a direct consequence of the first one and the fact that $A_3 \models \mathcal O$.
\end{proof}

At this point we managed to go from a model of $\mathcal M$ with carrier set $TX$ to a model of $\mathcal M$ with carrier set $STX$.

$$\underline{A_\mathcal{M}} = \underline{A_3} = \underline{A_2} = STX \qquad A_\mathcal{M} \brak{m \vdash x_{i_1} \cdots x_{i_n}} e = A_3\brak{\mu_n} (e \circ (j \mapsto i_j)) \qquad \eta_{A_2}^{A_\mathcal{M}} = \id$$

\medskip

\textbcf{Step 5.}

We have $A_\mathcal{M} \models \mathcal M$ and $A_2 \models \mathcal O \ast \mathcal A$. We can then use the pullback square to get a model $A := A_\mathcal{M} \triangleleft A_2$ of $\mathfrak T \triangleleft \mathcal A$:
\begin{itemize}
  \item $\underline{A} = \underline{A_\mathcal{M}} = \underline{A_2} = STX$
  \item $A \brak{(\cdot)}(x, y) = A_\mathcal{M} \brak{(\cdot)} (x, y) = A_2 \brak{\mu_2}(x, y)$
  \item $A \brak{(+)}(x, y) = A_2 \brak{(+)}(x, y)$
\end{itemize}

All axioms hold since they hold either in $A_\mathcal{M}$ or in $A_2$.

\medskip

We define $\eta : X \xrightarrow{\eta^{A_0}} \underline{A_0} = \underline{A_1} \xrightarrow{\eta_{A_1}^{A_2}} \underline{A_2} = STX = \underline{A}$.

\begin{proposition}
  $A$ is the free $\mathfrak T \triangleleft \mathcal A$-algebra over $X$.
\end{proposition}

\begin{proof}
  We need to show that $(A, \eta)$ is an universal arrow from $X$ to the forgetful functor $\mathcal{A} \to \Set$. Since the definitions for $(+)$ and $(\cdot)$ in $A$ are the same as we defined in Section~\ref{sec:comb1}, the same proof as in Section~\ref{prop:F_SGCSG_free} works here.
\end{proof}

\subsection{Counter example of Lemma~\ref{lem:A_M_ind_mod}}
\label{sec:counter_ex_lem}

\workingpoint

Let $\mathcal M = \BA$ be the theory of barycentric algebras, $\mathcal A = \SL$ be the theory of join-semilattices, $\OO$ be the simplex operad and $\mathfrak T : \OO \to \mathcal M$ be the usual translation sending a weight vector to the corresponding convex-combination term. \todo{to explicit}

\medskip

The free semilattice on a set $X$ is $F_\SL X = \mathcal P_f^+(X)$ the set of non-empty finite subsets of $X$, with union as join. \todo{to prove}

\medskip

The free barycentric algebra on $X$ is $F_\BA X = \mathcal D_f(X)$ the set of finitely-supported probability distributions on $X$. \todo{to prove}

\medskip

Let $X = \{x, y\}$. Following the pipeline of the previous section, let \begin{align*}
  A_0 &= F_\BA (X) = \mathcal D_f(X)\\
  A_1 &= \Mod(\mathfrak T, \Set) A_0\\
  A_2 &= \Mod(\OO, F_\SL) A_1
\end{align*}

The carrier set of $A_2$ is $$\ul{A_2} = \mathcal P_f^+ (\mathcal D_f (X))$$ the set of non-empty finite subsets of finitely supported probability distributions on $X$.

\medskip

For $p \in ]0, 1[$, let $f_p \in \OO_2$ be the binary convex-combination operation of weight $p$.

The lifted operation on $A_2$ is $$A_2 \brak{f_p} \la S, T\ra = \{p \mu + (1-p) \nu \mid \mu \in S,\ \nu \in T\}$$

Considering the barycentric axiom $x \oplus_p x = x$, it is of the form $$1 \vdash_\BA \mathfrak T f_p [\theta] = \mathfrak T \id$$ where \begin{itemize}
  \item $\theta : 2 \to 1$ is the constant map
  \item $\id \in \OO_1$ is the unary identity operation
\end{itemize}

Take $e := \{\delta_x,\; \delta_y\} \in \mathcal P_f^+ (\mathcal D_f (X))$.

$$A_2 \brak{f_p} \la e, e \ra = \{ \delta_x,\; p \delta_x + (1-p) \delta_y,\; p \delta_y + (1-p) \delta_x,\; \delta_y\}$$ whereas $$A_2 \brak{\id} e = e = \{\delta_x,\; \delta_y\}$$

Since $0 < p < 1$, $p \delta_x + (1-p) \delta_y$ and $p \delta_y + (1-p) \delta_x$ do not belong to $e$. Therefore $$A_2 \brak{f_p} \la e, e \ra \neq A_2 \brak{\id} e$$ and Lemma~\ref{lem:A_M_ind_mod} does not hold.

\subsection{General case}

Steps 1 to 4 of the previous section can be adapted to the general case. 

\medskip

$\eta^{A_0}$ comes from the adjunction $F_\mathcal{M} \dashv U : \Mod(\mathcal M, \Set) \to \Set$ and $\eta_{A_1}^{A_2}$ comes from the adjunction $\Mod(\mathcal O, F_\mathcal{A}) \dashv \Mod(\OO, \underline{-}) : \Mod(\OO, \Multi(\mathcal A, \Set)) \to \Mod(\OO, \la \Set ,\prod \ra)$.

\medskip

We encounter a problem when trying to generalize Lemma~\ref{lem:A_M_ind_mod}: it doesn't hold anymore (counter example of Section~\ref{sec:counter_ex_lem}). We need to replace it with another weaker lemma which limits which theories we can apply the construction to.

% ------------------------------------------------------------------------------------

\begin{definition}[$\mathfrak T$-regular presentation]
Let $\mathfrak T : \OO \to \mathcal{M}$ be a surjective translation.

A presentation of $\mathcal{M}$ over $\mathfrak T$ is \emph{$\mathfrak T$-regular} if:
\begin{itemize}
  \item for every operation symbol $g : n$ of the chosen presentation of $\mathcal{M}$, there is a chosen operation $\widetilde g \in \OO_n$ such that
  $$
  n \vdash_\mathcal{M} \mathfrak T\widetilde g = g(x_1,\dots,x_n),
  $$
  \item the chosen operations $\widetilde g$ generate $\OO$ under identities and operadic composition,
  \item every axiom of the presentation is of the form
  $$
  x_1,\dots,x_n \vdash_\mathcal{M} \mathfrak T f_1[\pi_1] = \mathfrak T f_2[\pi_2],
  $$
  for some $f_1,f_2 \in \OO_n$ and permutations $\pi_1,\pi_2 \in S_n$.
\end{itemize}
\end{definition}

\begin{lemma}[Step 4 in the $\mathfrak T$-regular case]
\label{lem:step4_correct}
With the preceding notation, assume that $\mathcal{M}$ admits a $\mathfrak T$-regular presentation. Let
$$
A_3 := \Mod(\OO,-)(A_2).
$$
For each operation symbol $g:n$ of the chosen presentation of $\mathcal{M}$, define
$$
A_\mathcal{M}\brak{g} := A_3\brak{\widetilde g}.
$$
Then:
\begin{itemize}
  \item for every $f \in \OO_n$, the interpretation in $A_\mathcal{M}$ of the $M$-term $\mathfrak Tf$ is equal to $A_3\brak{f}$;
  \item $A_\mathcal{M} \models M$.
\end{itemize}
\end{lemma}

\begin{proof}
Let $Y := \ul{A_1}$ and let $\eta_Y : Y \to \ul{F_\mathcal{A} Y}$ be the unit of the adjunction $F_\mathcal{A} \dashv U_\mathcal{A}$.

By construction of
$$
A_2 = \Mod(\OO,F_\mathcal{A})(A_1),
$$
for every $f \in \OO_n$, the operation $A_2\brak{f}$ is the unique multi-$A$-homomorphism extending
$$
Y^n \xrightarrow{A_1\brak{f}} Y \xrightarrow{\eta_Y} \ul{F_\mathcal{A} Y}.
$$

We first prove that, for every $f \in \OO_n$, the interpretation in $A_\mathcal{M}$ of the term $\mathfrak T f$ is equal to $A_3\brak{f}$. Since the chosen operations $\widetilde g$ generate $\OO$, we may prove this by induction on the operadic expression of $f$.

\begin{itemize}
  \item If $f = \mathrm{id}$, the claim is immediate.
  \item If $f = \widetilde g$ is one of the chosen generators, the claim is exactly the definition of $A_\mathcal{M}\brak{g}$.
  \item If $f = h \circ (f_1,\dots,f_k)$, then, using preservation of operadic composition by $\mathfrak T$, the induction hypothesis, and the fact that $A_3$ is an $\OO$-model, we get
  $$
  A_\mathcal{M}\brak{\mathfrak T f}
  = A_\mathcal{M}\brak{\mathfrak T h[\mathfrak T f_1,\dots,\mathfrak T f_k]}
  = A_3\brak{h} \circ (A_3\brak{f_1},\dots,A_3\brak{f_k})
  = A_3\brak{f}.
  $$
\end{itemize}

Now let
$$
x_1,\dots,x_n \vdash_\mathcal{M} \mathfrak T f_1[\pi_1] = \mathfrak T f_2[\pi_2]
$$
be an axiom of the chosen $\mathfrak T$-regular presentation. Since $A_0 \models \mathcal{M}$, we have equality of the corresponding operations on $Y$:
$$
A_1\brak{f_1} \circ \pi_1 = A_1\brak{f_2} \circ \pi_2.
$$
After postcomposing with $\eta_Y$, the resulting maps $Y^n \to \ul{F_\mathcal{A} Y}$ are still equal. By uniqueness of multi-$\mathcal A$-homomorphic extension, their extensions to $A_2$ are equal:
$$
A_2\brak{f_1} \circ \pi_1 = A_2\brak{f_2} \circ \pi_2.
$$
Forgetting from $A_2$ to $A_3$ preserves these equalities, so
$$
A_3\brak{f_1} \circ \pi_1 = A_3\brak{f_2} \circ \pi_2.
$$
By the first part of the proof, this means exactly that the two sides of the axiom have the same interpretation in $A_\mathcal{M}$. Hence all axioms of the chosen presentation hold in $A_\mathcal{M}$, and therefore $A_\mathcal{M} \models \mathcal{M}$.
\end{proof}

	\arrow["\lrcorner"{anchor=center, pos=0.125}, draw=none, from=2-2, to=0]
\end{tikzcd}\]

This gives us a unique morphism $h = H\, u : A \to B$ of $\mathfrak T \triangleleft \mathcal A$-algebras.

\medskip

We need to show that $h$ extends $f$ along $\eta$. $h_2 \circ \eta = h_4 \circ \eta = f$, therefore $h \circ \eta = f$ since $\eta$ comes from the pullback square.

\medskip

This morphism is unique for a given $h_2$ and $h_4$, but these are also uniquely determined by $h_0$ and $h_1$, which are themselves uniquely determined by $f$. Therefore $h$ is the unique morphism of $\mathfrak T \triangleleft \mathcal A$-algebras such that the following diagram commutes:
\[\begin{tikzcd}
  X \arrow[r, "\eta"] \arrow[dr, "f"'] & A \arrow[d, "h"]\\
  & B
\end{tikzcd}\]

\fbox{This proves that $A$ is the free $\mathfrak T \triangleleft \mathcal A$-algebra over $X$.}

\medskip

Now that we proved that $A$ is the free $\mathfrak T \triangleleft \mathcal A$-algebra over $X$, we can give an explicit description of the unique morphism $h : A \to B$ of $\mathfrak T \triangleleft \mathcal A$-algebras such that $h \circ \eta = f$.

\medskip

Since we cannot access the internal structure anymore, we have the following lemma:

\begin{lemma}
  \label{lem:decom_A2}
  For every $e \in \underline{A_2}$, there is some $s \in \mathcal A_k$ and $\ebf \in \ul{A_1}^k$ such that $$e = A_2 \brak s \la \eta_{A_1}^{A_2} \ebf_i \ra_{i=1}^k$$
\end{lemma}

\begin{proof}
  This is proved in Ohad's notes.
\end{proof}

We can actually prove a stronger statement:

\begin{lemma}
  \label{lem:decomp_A2_strong}
  For all $e \in \ul{A_2}$, there exists $s \in \mathcal A_k$ such that for all $i \in [k]$, there exists $t_i \in \mathcal M_{k_i}$ and $\ebf_{\la i, j \ra} \in X$ such that:
  $$e = A_2 \brak s \la \eta_{A_1}^{A_2} A_0 \brak{t_i} \la \eta^{A_1} \ebf_{\la i, j \ra} \ra_{j=1}^{k_i} \ra_{i=1}^k$$
\end{lemma}

\begin{proof}
  Similar to the proof of Lemma~\ref{lem:decom_A2}.
\end{proof}

Let $e \in \underline{A} = \underline{A_2}$. By Lemma~\ref{lem:decomp_A2_strong}, we can write $$e = A_2 \brak s \la \eta_{A_1}^{A_2} A_0 \brak{t_i} \la \eta^{A_1} \ebf_{\la i, j \ra} \ra_{j=1}^{k_i} \ra_{i=1}^k$$

We can then define $h : X \to \underline{B}$ by $$h\, e = B \brak{s} \la B \brak{t_i} \la f\,\ebf_{\la i,j \ra} \ra_{j=1}^{k_i} \ra_{i=1}^k$$

\begin{lemma}
  $h$ is well-defined, i.e. it does not depend on the choice of $s, t_i$ and $\ebf_{\la i, j \ra}$.
\end{lemma}

\begin{proof}
  Suppose $$e = A_2 \brak{s_1} \la \eta_{A_1}^{A_2} A_0 \brak{t_{1, k}} \la \eta^{A_1} \ebf_{\la 1, k, j \ra} \ra_{j=1}^{k_{1, k}} \ra_{k=1}^{n_1}$$ and $$e = A_2 \brak{s_2} \la \eta_{A_1}^{A_2} A_0 \brak{t_{2, k}} \la \eta^{A_1} \ebf_{\la 2, k, j \ra} \ra_{j=1}^{k_{2, k}} \ra_{k=1}^{n_2}$$

  Equalities in $A$ are exactly generated by the equations of $\mathfrak T \triangleleft \mathcal A$.
  
  Since $B$ is a model of $\mathfrak T \triangleleft \mathcal A$, we have that $B$ validates the same equations, and therefore
  \begin{align*}
    &B \brak{s_1} \la B \brak{t_{1, k}} \la f\,\ebf_{\la 1, k, j \ra} \ra_{j=1}^{k_{1, k}} \ra_{k=1}^{n_1}\\
    =& B \brak{s_2} \la B \brak{t_{2, k}} \la f\,\ebf_{\la 2, k, j \ra} \ra_{j=1}^{k_{2, k}} \ra_{k=1}^{n_2}
  \end{align*}
\end{proof}

We now prove that this is indeed a $\mathfrak T \triangleleft \mathcal A$-homomorphism and that it is the unique one such that the following diagram commutes:
\[\begin{tikzcd}
  X & {\underline{A}} \\
  & {\underline{B}}
  \arrow["\eta", from=1-1, to=1-2]
  \arrow["f"', from=1-1, to=2-2]
  \arrow["h", from=1-2, to=2-2]
\end{tikzcd}\]

\begin{itemize}[label=$\triangleright$]
  \item \textbf{preservation of $\mathcal A$-structure.}
  
  Take any $s \in \mathcal A_n$ and $e_1, \dots, e_n \in \underline{A}$. We can write $$e_l = A_2 \brak{s^l} \la \eta_{A_1}^{A_2} A_0 \brak{t_i^l} \la \eta^{A_1} \ebf_{\la i, j \ra}^l \ra_{j=1}^{k_i^l} \ra_{i=1}^{k^l}$$
  Taking $p := \sum_l k^l$ and $s' := s [i \mapsto s^i]_{i=1}^n \in \mathcal A_p$, we have:
  \begin{align*}
    h \lp A_2 \brak{s} \la e_l \ra_{l=1}^n \rp &= h \lp A_2 \brak{s} \la A_2 \brak{s^l} \la \eta_{A_1}^{A_2} A_0 \brak{t_i^l} \la \eta^{A_1} \ebf_{\la i, j \ra}^l \ra_{j=1}^{k_i^l} \ra_{i=1}^{k^l} \ra_{l=1}^n \rp\\
    &= h \lp A_2 \brak{s'} \la \eta_{A_1}^{A_2} A_0 \brak{t_m} \la \eta^{A_1} \ebf_j^m \ra_{j=1}^{k_m} \ra_{m=1}^p \rp\\
    &= B \brak{s'} \la B \brak{t_m} \la f\,\ebf_j^m \ra_{j=1}^{k_m} \ra_{m=1}^p\\
    &= B \brak{s} \la B \brak{s^l} \la B \brak{t_i^l} \la f\,\ebf_{\la i, j \ra}^l \ra_{j=1}^{k_i^l} \ra_{i=1}^{k^l} \ra_{l=1}^n\\
    &= B \brak{s} \la h\, e_l \ra_{l=1}^n
  \end{align*}


  \item \textbf{preservation of $\mathcal M$-structure.}
  
  Take any $t \in \mathcal M_n$ and $e_1, \dots, e_n \in \underline{A}$. We can write $$e_l = A_2 \brak{s^l} \la \eta_{A_1}^{A_2} A_0 \brak{t_i^l} \la \eta^{A_1} \ebf_{\la i, j \ra}^l \ra_{j=1}^{k_i^l} \ra_{i=1}^{k^l}$$
  Since the operations of $\mathcal M$ distribute over the operations of $\mathcal A$, we have:
  \begin{align*}
    \hspace{-2cm} h \lp A_2 \brak{t} \la e_l \ra_{l=1}^n \rp &= h \lp A_2 \brak{t} \la A_2 \brak{s^l} \la \eta_{A_1}^{A_2} A_0 \brak{t_i^l} \la \eta^{A_1} \ebf_{\la i, j \ra}^l \ra_{j=1}^{k_i^l} \ra_{i=1}^{k^l} \ra_{l=1}^n \rp\\
    &= h \lp A_2 \brak{s^1} \la \cdots A_2 \brak{s^n} \la A_2 \brak{t} \la \eta_{A_1}^{A_2} A_0 \brak{t_{i_l}^l} \la \eta^{A_0} \ebf_{\la i_l, m \ra}^l \ra_{m=1}^{k_{i_l}^l} \ra_{l=1}^n \ra_{i_n = 1}^{k^n} \cdots \ra_{i_1 = 1}^{k^1} \rp\\ 
    &= h \lp A_2 \brak{s^1 \la \cdots s^n \la x_{i_1}, \dots, x_{i_n} \ra_{i_n = 1}^{k^n} \cdots \ra_{i_1 = 1}^{k^1}} \la A_2 \brak{t} \la \eta_{A_1}^{A_2} A_0 \brak{t_{\la q \ra_l^{-1}}^l} \la \eta^{A_0} \ebf_{\la \la q \ra_l^{-1}, m \ra}^l \ra_{m=1}^{k_{\la q \ra_l^{-1}}^l} \ra_{l=1}^n \ra_{q=1}^{\prod_{l=1}^n k^l} \rp\\
    &= h \lp A_2 \brak{s^1 \la \cdots s^n \la x_{i_1}, \dots, x_{i_n} \ra_{i_n = 1}^{k^n} \cdots \ra_{i_1 = 1}^{k^1}} \la \eta_{A_1}^{A_2} A_0 \brak{t'_{q}} \la \eta^{A_0} \ebf^m \ra_{m=1}^{\sum_l k_{\la q \ra_l^{-1}}^l} \ra_{q=1}^{\prod_{l=1}^n k^l} \rp\\
    &= B \brak{s^1 \la \cdots s^n \la x_{i_1}, \dots, x_{i_n} \ra_{i_n = 1}^{k^n} \cdots \ra_{i_1 = 1}^{k^1}} \la B \brak{t'_{q}} \la f\,\ebf^m \ra_{m=1}^{\sum_l k_{\la q \ra_l^{-1}}^l} \ra_{q=1}^{\prod_{l=1}^n k^l}\\
    &= B \brak{s^1 \la \cdots s^n \la x_{i_1}, \dots, x_{i_n} \ra_{i_n = 1}^{k^n} \cdots \ra_{i_1 = 1}^{k^1}} \la B \brak{t} \la B \brak{t_{\la q \ra_l^{-1}}^l} \la f\,\ebf_{\la \la q \ra_l^{-1}, m \ra}^l \ra_{m=1}^{k_{\la q \ra_l^{-1}}^l} \ra_{l=1}^n \ra_{q=1}^{\prod_{l=1}^n k^l}\\
    &= B \brak{s^1} \la \cdots B \brak{s^n} \la B \brak{t} \la B \brak{t_{i_l}^l} \la f\, \ebf_{\la i_l, m \ra}^l \ra_{m=1}^{k_{i_l}^l} \ra_{l=1}^n \ra_{i_n = 1}^{k^n} \cdots \ra_{i_1 = 1}^{k^1}\\
    &= B \brak{t} \la B \brak{s^l} \la B \brak{t_i^l} \la f\,\ebf_{\la i, j \ra}^l \ra_{j=1}^{k_i^l} \ra_{i=1}^{k^l} \ra_{l=1}^n\\
    &= B \brak{t} \la h\, e_l \ra_{l=1}^n
  \end{align*}

  Note to self: never again.

  \item \textbf{commutation of the diagram.}
  
  We need to show that for all $x \in X$, we have $h(\eta\, x) = f\, x$. We have $\eta x = \eta_{A_1}^{A_2} \eta^{A_0}\, x$. We notice that
  \begin{align*}
    \eta\, x &= \eta_{A_1}^{A_2} \eta^{A_0}\, x\\
    &= \eta_{A_1}^{A_2} A_0 \brak{\id} \la \eta^{A_1}\, x \ra\\
    &= A_2 \brak{\id} \la \eta_{A_1}^{A_2} A_0 \brak{\id} \la \eta^{A_1}\, x \ra \ra
  \end{align*}
  Therefore
  \begin{align*}
    h(\eta\, x) &= h \lp A_2 \brak{\id} \la \eta_{A_1}^{A_2} A_0 \brak{\id} \la \eta^{A_1}\, x \ra \ra \rp\\
    &= B \brak{\id} \la B \brak{\id} \la f\, x \ra \ra\\
    &= f\, x
  \end{align*}
  
  \item \textbf{uniqueness.}
  
  Let $h' : A \to B$ be another morphism of $\mathfrak T \triangleleft \mathcal A$-algebras such that $h' \circ \eta = f$. We need to show that for all $e \in \underline{A}$, we have $h\, e = h'\, e$. We can write $$e = A_2 \brak s \la \eta_{A_1}^{A_2} A_0 \brak{t_i} \la \eta^{A_1} \ebf_{\la i, j \ra} \ra_{j=1}^{k_i} \ra_{i=1}^k$$
  Therefore
  \begin{align*}
    h\, e &= B \brak{s} \la B \brak{t_i} \la f\,\ebf_{\la i,j \ra} \ra_{j=1}^{k_i} \ra_{i=1}^k &\text{by definition}\\
    &= B \brak{s} \la B \brak{t_i} \la  h'\eta\, \ebf_{\la i,j \ra} \ra_{j=1}^{k_i} \ra_{i=1}^k &\text{since the diagram commutes}\\
    &= h' \lp A_2 \brak{s} \la \eta_{A_1}^{A_2} A_0 \brak{t_i} \la \eta^{A_1} \ebf_{\la i, j \ra} \ra_{j=1}^{k_i} \ra_{i=1}^k \rp &\text{since } h' \text{ is an homomorphism}\\
    &= h'\, e
  \end{align*}
  and $h = h'$.
\end{itemize}

\fbox{Therefore $h$ is the unique morphism of $\mathfrak T \triangleleft \mathcal A$-algebras such that $h \circ \eta = f$.}


\section{Constructing free extensions}

\subsection{Semigroups}
\label{sec:free_extension_semigroups}

\minisection{Semigroup}

\medskip

Let's construct the free extension of an $\SG$-model $A = (\ul{A}, +)$ by a set of variable $X$.

\medskip

We define a $\SG$-algebra $\Frex_A(X)$ as follows:
\begin{itemize}
  \item The carrier set of $\Frex_A(X)$ is the set of non-empty alternating lists of elements of $A$ and $X^+$.
  \item The operator $\cdot$ on $\Frex_A(X)$ is defined by concatenation of lists, and by using the operator $+$ of $A$ to combine adjacent elements of $A$ in the resulting list and list concatenation to combine adjacent elements of $X^+$.
\end{itemize}

This is indeed an $\SG$-aglebra, as list concatenation is associative, and the operator $+$ of $A$ is associative.

\medskip

We define $\embed : A \to \Frex_A(X)$ by $\embed(n) = [n]$ for all $n \in A$. For all $m, n \in A$, we have $\embed(m + n) = [m + n] = [m] \cdot [n] = \embed(m) \cdot \embed(n)$, so $\embed$ is a morphism of $\SG$-algebras.

\medskip

We also define $\var : X \to \Frex_A(X)$ by $\var(x) = [x]$ for all $x \in X$.

\medskip

\begin{proposition}
  $\la \Frex_A(X), \embed, \var\ra$ is the free extension of $A$ by $X$.
\end{proposition}

\begin{proof}
  Suppose $\la B, f, g\ra$ is another extension of $A$ by $X$. 
  
  We extend $g$ to $X^+$ by defining $g(x_1 \cdots x_n) = g(x_1) \cdots g(x_n)$ for all $x_1, \dots, x_n \in X$. 
  
  Now for an element $[b_1, \dots, b_n]$ of $\Frex_A(X)$, we define $h([b_1, \dots, b_n]) = \brak{b_1} \cdot \cdots \cdot \brak{b_n}$ where:
  \begin{itemize}
    \item if $b_i \in A$, then $\brak{b_i} = f(b_i)$
    \item if $b_i \in X^+$, then $\brak{b_i} = g(b_i)$
  \end{itemize}
  This is well-defined and homomorphic because the only merges in the extension are:
  \begin{itemize}
    \item concrete/concrete, handled by $f(a+b) = f(a) \cdot f(b)$
    \item abstract/abstract, handled by $g(uv) = g(u) \cdot g(v)$
  \end{itemize}
  The diagram below commutes:
\[
\begin{tikzcd}
  A \arrow[r, "\embed"] \arrow[dr, "f"'] & \Frex_A(X) \arrow[d, "h"] & X \arrow[l, "\var"'] \arrow[dl, "g"] \\
  & B &
\end{tikzcd}
\]
  Uniqueness of $h$ is straightforward, as $h$ is defined by the values of $f$ and $g$ on the elements of $A$ and $X$, and by the homomorphic property of $h$.
\end{proof}

\minisection{Commutative semigroup}

\medskip

Let's construct the free extension of an $\SG$-model $A = (\ul{A}, +)$ by a set of variable $X$. It is going to be linear polynomials in $A[X]$.

\medskip

We define a $\SG$-algebra $\Frex_A(X)$ as follows:
\begin{itemize}
  \item The carrier set of $\Frex_A(X)$ is the set $\{c + a_1 x_1 + \cdots + a_n x_n \mid n \ge 0, c \in A, x_1, \dots, x_n \in X\}$ \tocheck{TO REVISE: there can be no constant part (currently more suited for commutative monoids)}
  \item The operator $+$ on $\Frex_A(X)$ is defined by $$(c + a_1 x_1 + \cdots + a_n x_n) + (d + b_1 y_1 + \cdots + b_m y_m) = (c + d) + \sum_{z_i \in \{x_1, \dots, x_n \} \cup \{y_1, \dots, y_m\}}c_i z_i$$ where $c_i$ is the sum of the coefficients of $z_i$ in the two polynomials.
\end{itemize}

This is indeed an $\SG$-algebra, as addition of polynomials is associative and commutative.

\medskip

We define $\embed : A \to \Frex_A(X)$ by $\embed(n) = n$ for all $n \in A$. For all $m, n \in A$, we have $\embed(m + n) = m + n = \embed(m) + \embed(n)$, so $\embed$ is a morphism of $\SG$-algebras.

\medskip

We also define $\var : X \to \Frex_A(X)$ by $\var(x) = x$ for all $x \in X$.

\begin{proposition}
  $\la \Frex_A(X), \embed, \var\ra$ is the free extension of $A$ by $X$.
\end{proposition}

\begin{proof}
  Suppose $\la B, f, g\ra$ is another extension of $A$ by $X$. We define $h : \Frex_A(X) \to B$ by $$h(c + a_1 x_1 + \cdots + a_n x_n) = f(c) + \sum_{i=1}^n f(a_i) \cdot g(x_i)$$

  This is well-defined and homomorphic because the only merges in the extension are:
  \begin{itemize}
    \item concrete/concrete, handled by $f(a+b) = f(a) + f(b)$
  \end{itemize}

  The diagram below commutes:
\[\begin{tikzcd}
  A \arrow[r, "\embed"] \arrow[dr, "f"'] & \Frex_A(X) \arrow[d, "h"] & X \arrow[l, "\var"'] \arrow[dl, "g"] \\
  & B &
\end{tikzcd}\]
  Uniqueness of $h$ is straightforward, as $h$ is defined by the values of $f$ and $g$ on the elements of $A$ and $X$, and by the homomorphic property of $h$.
\end{proof}

\subsection{Sem-rgs}

Let $A = (\ul{A}, (+), (\cdot))$ be a $\SGCSG$-model and $X$ be a set.

\medskip

We define $$\Mon_A(X) = \set{ [b_1, \dots, b_n] \mid n \ge 1,\ b_i \in X^+ \sqcup \ul{A},\ b_ib_{i+1}\text{ alternating kinds},\ \exists i,b_i \in X^+}$$

We pose $A_\bot = \ul{A} \sqcup \{\bot\}$ where $\bot$ is a new element, designating the absence of a concrete part in the polynomial. We extend the operators $+$ and $\cdot$ of $A$ to $A_\bot$ by defining $\bot + a = a + \bot = a$, $\bot \cdot a = a \cdot \bot = \bot$ for all $a \in A$ and $\bot + \bot = \bot$, $\bot \cdot \bot = \bot$.

\medskip

We define $$\Frex_A(X) = \set{ (c, M) \mid c \in A_\bot,\ M \in \Bag(\Mon_A(X)),\ c \neq \bot \text{ or } M \neq \varnothing }$$

\medskip

Defining the addition is straightforward: $$(c, M) + (d, N) = (c + d, M \uplus N)$$

We define the monomial multiplication $m \otimes n$ for $m, n \in \Mon_A(X)$ by concatenation of lists, and by using the operator $\cdot$ of $A$ to combine adjacent elements of $A$ in the resulting list and list concatenation to combine adjacent elements of $X^+$ (as in Section~\ref{sec:free_extension_semigroups}).

\medskip

We also define multiplication of a concrete element with a monomial by treating the concrete element as a monomial of length 1:
\begin{itemize}
  \item $r \triangleright m = [r] \otimes m$
  \item $m \triangleleft r = m \otimes [r]$
\end{itemize}
This is well-defined because $m$ already contains an element of $X^+$.

\medskip

We naturaly extend the multiplication to bags, which then gives us a definition for the multiplication of two elements of $\Frex_A(X)$:
$$(c, M) \cdot (d, N) = (c \cdot d, (c \triangleright N) \uplus (M \otimes N) \uplus (M \triangleleft d))$$ 

\medskip

We define $\embed : A \to \Frex_A(X)$ by $\embed(a) = (a, \varnothing)$ for all $a \in A$. For all $a, b \in A$, we have $\embed(a + b) = (a + b, \varnothing) = (a, \varnothing) + (b, \varnothing) = \embed(a) + \embed(b)$ and $\embed(a \cdot b) = (a \cdot b, \varnothing) = (a, \varnothing) \cdot (b, \varnothing) = \embed(a) \cdot \embed(b)$, so $\embed$ is a morphism of $\SGCSG$-algebras.

\medskip

We also define $\var : X \to \Frex_A(X)$ by $\var(x) = (\bot, \{[x]\})$ for all $x \in X$.

\begin{proposition}
  $\la \Frex_A(X), \embed, \var\ra$ is the free extension of $A$ by $X$.
\end{proposition}

\begin{proof}
  \begin{itemize}
    \item \textbf{$\Frex_A(X)$ is a $\SGCSG$-model.} All axioms of $\SGCSG$ but the distributivity axioms are straightforward to check. For the distributivity axioms, we have:
\begin{align*}
  (c, M) \cdot ((d, N) + (e, P)) &= (c, M) \cdot (d + e, N \uplus P)\\
  &= (c \cdot (d + e), (c \triangleright (N \uplus P)) \uplus (M \otimes (N \uplus P)) \uplus (M \triangleleft (d + e)))\\
  &= (c \cdot d + c \cdot e, (c \triangleright N) \uplus (c \triangleright P) \uplus (M \otimes N) \uplus (M \otimes P) \uplus (M \triangleleft d) \uplus (M \triangleleft e))\\
  &= (c, M) \cdot (d, N) + (c, M) \cdot (e, P)
\end{align*}
and
\begin{align*}
  ((c, M) + (d, N)) \cdot (e, P) &= (c + d, M \uplus N) \cdot (e, P)\\
  &= ((c + d) \cdot e, ((c + d) \triangleright P) \uplus ((M \uplus N) \otimes P \uplus ((M \uplus N) \triangleleft e)))\\
  &= (c \cdot e + d \cdot e, (c \triangleright P) \uplus (d \triangleright P) \uplus (M \otimes P) \uplus (N \otimes P) \uplus (M \triangleleft e) \uplus (N \triangleleft e))\\
  &= (c, M) \cdot (e, P) + (d, N) \cdot (e, P)
\end{align*}
  \item \textbf{Universality.} Let $B$ be any $\SGCSG$-model and let $f : A \to B$ and $g : X \to B$ be morphisms of $\SGCSG$-algebras. 
  
  We extend $g$ to $X^+$ by defining $g(x_1 \cdots x_n) = g(x_1) \cdots g(x_n)$ for all $x_1, \dots, x_n \in X$.

  We define $h([b_1, \dots, b_n]) = \brak{b_1} \cdot \cdots \cdot \brak{b_n}$ where:
  \begin{itemize}
    \item if $b_i \in A$, then $\brak{b_i} = f(b_i)$
    \item if $b_i \in X^+$, then $\brak{b_i} = g(b_i)$
  \end{itemize}
  We extend $h$ to $\Frex_A(X)$ by defining $$h(c, M) = \begin{cases}
    f(c) + \sum_{m \in M} h(m) &\text{ if } c \neq \bot\\
    \sum_{m \in M} h(m) &\text{ if } c = \bot
  \end{cases}$$

  This is well-defined and homomorphic\footnote{it is less straightforward than in the examples for the semigroups, so doing it may be a good idea} because the only merges in the extension are:
  \begin{itemize}
    \item concrete/concrete, handled by $f(a+b) = f(a) + f(b)$ and $f(a \cdot b) = f(a) \cdot f(b)$
    \item abstract/abstract, handled by $g(uv) = g(u) \cdot g(v)$
  \end{itemize}
  The diagram below commutes:
\[\begin{tikzcd}
  A \arrow[r, "\embed"] \arrow[dr, "f"'] & \Frex_A(X) \arrow[d, "h"] & X \arrow[l, "\var"'] \arrow[dl, "g"] \\
  & B &
\end{tikzcd}\]
  Uniqueness of $h$ is straightforward, as $h$ is defined by the values of $f$ and $g$ on the elements of $A$ and $X$, and by the homomorphic property of $h$.
\end{itemize}
\end{proof}


\subsection{Distributive combination of free extensions}

We'll try to adapt the construction of Section~\ref{sec:free_alg_ex} to the case of free extensions.

\medskip

Let $B$ be a $\mathfrak T \triangleleft \mathcal A$ model, and $X$ be a set.

\medskip

\textbcf{Step 1.} 

Let $B_\mathcal M$ be the underlying $\mathcal M$-model of $B$. Let $E_0 := \Frex_{B_\mathcal{M}}(X) \models \mathcal M$ be the free extension of $B_\mathcal{M}$ by $X$. From the free extension construction, we also have the canonical inclusion $i_\mathcal{M} : B_\mathcal{M} \to E_0$.

\medskip

\textbcf{Step 2.} 

We apply the functor $$\underline{-}_\mathfrak{T} := \Mod(\mathfrak T, \Set) : \Mod(\mathcal M, \Set) \to \Mod(\mathcal O, \la \Set, \prod \ra)$$ to $E_0$. This gives $E_1 := \underline{E_0}_\mathfrak{T} \models \mathcal O$, and the canonical inclusion $i_\mathcal{O} : B_\mathcal{O} \to E_1$.

\medskip

\textbcf{Step 3.}

Now the next part is tricky. Let's first look at this abstractly, and then find a way to actually construct the required elements.

\medskip

From \cite{Allais2025}, we know that we can construct free extensions based on free algebras. This means that there exists $E_1^{\mathrm{new}}$ such that $E_1 \cong B_\mathcal{O} \coprod E_1^{\mathrm{new}}$. The model of $\OO \triangleleft \mathcal A$ that we want is then $$E_2 = B_{\OO \triangleleft \mathcal A} \coprod F_{\mathcal O \to \mathcal A} E_1^{\mathrm{new}}$$ with $$F_{\mathcal O \to \mathcal A} : \Mod(\mathcal O, F_\mathcal{A}) : \Mod(\mathcal O, \la \Set, \prod \ra) \to \Mod (\mathcal O, \Multi(\mathcal A, \Set))$$ obtained as in step 3 of Section~\ref{sec:free_alg_ex}. The problem with this is that we can't get $E_1^{\mathrm{new}}$ from $E_1$, so we'll have to find another approach.

\medskip

Let's note $F_{\OO \to \mathcal A} \dashv U_{\OO \to \mathcal A}$ the adjunction between $\Mod(\mathcal O, \la \Set, \prod \ra)$ and $\Mod (\mathcal O, \Multi(\mathcal A, \Set))$. We have $B_\OO = U_{\OO \to \mathcal A} B_{\OO \triangleleft \mathcal A}$. Therefore the identity $\id_{B_\OO} : B_\OO \to U_{\OO \to \mathcal A} B_{\OO \triangleleft \mathcal A}$ has an adjunct $\epsilon_B : F_{\OO \to \mathcal A} B_\OO \to B_{\OO \triangleleft \mathcal A}$. We can then define $E_2$ as the pushout of $\epsilon_B$ and $F_{\OO \to \mathcal A} i_\mathcal{O}$:
\[\begin{tikzcd}
	{F_{\OO \to \mathcal A} B_\OO} & {F_{\OO \to \mathcal A}E_1} \\
	{B_{\OO \triangleleft \mathcal A}} & {E_2}
	\arrow["{F_{\OO \to \mathcal A}i_0}", from=1-1, to=1-2]
	\arrow["{\epsilon_B}"', from=1-1, to=2-1]
	\arrow[from=1-2, to=2-2]
	\arrow[from=2-1, to=2-2]
\end{tikzcd}\]

\begin{lemma}
  \label{lem:pushout_coprod_inj}
  Let $\mathcal C$ be a category, $X, Y, Z$ be objects of $\mathcal C$, and $f : X \to Z$. 
  
  The pushout of $\iota_X : X \to X \coprod Y$ and $f$ is isomorphic to $Y \coprod Z$: $$(X \coprod Y) \amalg_X Z \cong Y \coprod Z$$
\end{lemma}

\begin{proof}
  % This is Theorem 2.2 of \cite{ITA_2005__39_3_511_0}.
%   We have the following pushout square:
%   \[\begin{tikzcd}
%   X & {X \coprod Y} \\
%   Z & P
%   \arrow["\iota_X", from=1-1, to=1-2]
%   \arrow["f"', from=1-1, to=2-1]
%   \arrow["u",from=1-2, to=2-2]
%   \arrow["v"', from=2-1, to=2-2]
% \end{tikzcd}\]
%   We want to show that $P$ is isomorphic to $Y \coprod Z$. Define $w : [v, u \circ \iota_Y] : Y \coprod Z \to P$.
\end{proof}

\begin{lemma}
  $$E_2 := B_{\OO \triangleleft A} \coprod_{F_{\OO \to \mathcal A} B_\OO} F_{\OO \to \mathcal A} E_1 \cong B_{\OO \triangleleft \mathcal A} \coprod F_{\OO \to \mathcal A} E_1^{\mathrm{new}}$$
\end{lemma}

\begin{proof}
  Because $F_{\OO \to \mathcal A}$ is a left adjoint, it preserves coproducts, so $$F_{\OO \to \mathcal A} E_1 \cong F_{\OO \to \mathcal A} B_\OO \coprod F_{\OO \to \mathcal A} E_1^{\mathrm{new}}$$ $F_{\OO \to \mathcal A} i_\OO : F_{\OO \to \mathcal A} B_\OO \to F_{\OO \to \mathcal A} E_1$ is then the first coproduct injection. Therefore the pushout span for $E_2$ is $$B_{\OO \to \mathcal A} \xleftarrow{\epsilon_B} F_{\OO \to \mathcal A} B_\OO \xrightarrow{\iota_1} F_{\OO \to \mathcal A} B_\OO \coprod F_{\OO \to \mathcal A} E_1^{\mathrm{new}}$$
  By Lemma~\ref{lem:pushout_coprod_inj}, the pushout of this span is isomorphic to the coproduct $B_{\OO \triangleleft \mathcal A} \coprod F_{\OO \to \mathcal A} E_1^{\mathrm{new}}$.
\end{proof}

Since the pushout is taken in $\Mod(\OO, \Multi(\mathcal A, \Set))$, $E_2$ is a model of $\OO$ in $\Multi(\mathcal A, \Set)$ and we have the canonical inclusion $B_{\OO \triangleleft \mathcal A} \to E_2$.

\medskip

\textbf{Intuition:} The intuition behind this construction is that we can't directly apply $F_{\OO \to \mathcal A}$ to $E_1$ because the concrete part is then duplicated. By doing the pushout, we separate the concrete part (with $\epsilon_B$) from the new part (with $F_{\OO \to \mathcal A} i_\OO$), and we get the desired result.

\medskip

\textbcf{Step 4.}

Let $E_3 := \Mod(\OO, \underline{-}) E_2 \models \OO$ be the underlying operadic algebra. We use surjectivity as in step 4 of Section~\ref{sec:free_alg_ex} to get an $\mathcal M$-model $E_\mathcal{M} \models \mathcal M$.

\medskip

\textbcf{Step 5.}

We can the define $E$ as the pullback of $E_\mathcal{M}$ and $E_2$. This should give us the free extension of $B$ by $X$.
